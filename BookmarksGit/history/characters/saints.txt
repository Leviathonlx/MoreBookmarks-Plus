# Virgin Mary
saint_0001 = {
	name = "Mary"
	female = yes
	martial = 3
	diplomacy = 5
	intrigue = 5
	stewardship = 10
	prowess = 3
	rite = "roman_rite"
	obscured = yes

	culture = "hebrew"
	trait = education_diplomacy_3
	trait = chaste
	trait = compassionate
	trait = forgiving
	-17.1.1 = {
		birth = yes
	}
	1.1.1 = {
		give_nickname = nick_the_virgin
	}
	# Baronius in his Annals relies on a passage in the Chronicon of Eusebius for his assumption that Mary died A.D. 48.
	48.1.1 = {
		death = yes
		effect_even_if_dead = {
			province:5965 = {
				set_variable = hide_location_for_saint
			}
			pam_canonize_saint_effect = {	
				CANONIZER = root
				CANDIDATE = root
				CONSECRATION_PROVINCE = province:5965 #Jerusalem
				PRESET = yes
			}
			religion:christianity_religion = {
				every_faith = {
					limit = {
						main_rite = { is_rite_enabled = yes }
						NOT = { has_saint_character = root }
					}
					add_saint = { character = root }
				}
			}
			# looks strange as we don't have any other muslim saints preset yet
			#religion:islamic_religion = {
			#	every_faith = {
			#		limit = {
			#			main_rite = { is_rite_enabled = yes }
			#			NOT = { has_saint_character = root }
			#		}
			#		add_saint = { character = root }
			#	}
			#}
			if = {
				limit = {
					NOT = {
						title:c_jerusalem.holder.faith = {
							has_saint_character = character:saint_0001
						}
					}
				}
				title:c_jerusalem.holder.faith = {
					add_saint = { character = character:saint_0001 }
				}
			}
			title:c_jerusalem.holder = {
				consume_saint_relic_charge = character:saint_0001
				consume_saint_relic_charge = character:saint_0001
				consume_saint_relic_charge = character:saint_0001
				consume_saint_relic_charge = character:saint_0001 #Assumption of Mary, avoid some random blasphemy against our beloved mother
			}
			if = {
				limit = {
					title:c_jerusalem.holder.faith = {
						NOT = { religion = religion:christianity_religion }
					}
				}
				title:c_jerusalem.holder.faith = {
					remove_saint = character:saint_0001
				}
			}
		}
	}	
}
