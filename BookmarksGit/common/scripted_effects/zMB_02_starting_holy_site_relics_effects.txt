# Overrides of effects from 02_starting_holy_site_relics_effects.txt

create_starting_relic_constantinople_christian_effect = {
	# After 1204 the Crown of Thorns is held by the Latin emperors and, from 1239, by the kings of France (historical_artifacts.0023)
	if = {
		limit = { current_date < 1204.4.12 }
		set_artifact_rarity_illustrious = yes
		scope:relic_owner = {
			create_artifact = {
				# The Mandylion already exists as the Image of Edessa
				name = artifact_relic_crown_of_thorns_name
				description = artifact_relic_crown_of_thorns_desc
				type = pedestal
				template = christian_artifact_template
				visuals = pedestal_christian_relic
				wealth = scope:wealth
				quality = scope:quality
				history = { type = created_before_history }
				modifier = artifact_placeholder_modifier
				save_scope_as = newly_created_artifact
			}
		}
		# Sets christian_crown so historical_artifacts.0023 will not pick another Crown of Thorns
		scope:newly_created_artifact = { set_variable = christian_crown }
		finalize_starting_unique_relic_effect = yes
	}

	set_artifact_rarity_illustrious = yes
	scope:relic_owner = {
		create_artifact = {
			name = artifact_relic_baptist_arm_name
			description = artifact_relic_baptist_arm_desc
			type = pedestal
			template = christian_artifact_template
			visuals = pedestal_christian_relic
			wealth = scope:wealth
			quality = scope:quality
			history = { type = created_before_history }
			modifier = artifact_placeholder_modifier
			save_scope_as = newly_created_artifact
		}
	}
	finalize_starting_unique_relic_effect = yes
}

create_starting_relic_mecca_islamic_effect = {
	# The Qarmatians of Bahrain carried the Black Stone off to Hajar in 930 and returned it in 951
	if = {
		limit = {
			current_date >= 930.1.12
			current_date < 951.1.1
		}
		# Mecca is an eminent site of several faiths, so only create the stone once
		if = {
			limit = { any_artifact = { has_variable = MB_black_stone_qarmatian } }
		}
		else_if = {
			limit = { exists = title:k_bahrein.holder }
			title:k_bahrein.holder = {
				create_artifact_MB_black_stone_qarmatian_effect = { OWNER = this }
			}
		}
		else_if = {
			limit = { exists = title:d_al-hasa.holder }
			title:d_al-hasa.holder = {
				create_artifact_MB_black_stone_qarmatian_effect = { OWNER = this }
			}
		}
	}
	else = {
		set_artifact_rarity_illustrious = yes
		scope:relic_owner = {
			create_artifact = {
				name = artifact_relic_black_stone_name
				description = artifact_relic_black_stone_desc
				type = pedestal
				template = islam_artifact_template
				visuals = pedestal_islamic_relic
				wealth = scope:wealth
				quality = scope:quality
				history = { type = created_before_history }
				modifier = artifact_placeholder_modifier
				save_scope_as = newly_created_artifact
			}
		}
		finalize_starting_unique_relic_effect = yes
	}
}
