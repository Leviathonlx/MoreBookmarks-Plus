# Prince-electors of the Holy Roman Empire by period. Used by game_start.txt (HRE SETUP) and the 1198 change-over in z_MB_vanilla_compat_on_actions.txt.

# MB Added

# Before 1198 there was no fixed college: kings were chosen by the great princes, led by the Rhenish archbishops and the dukes of the stem duchies.
# Lothair III was elected in 1125 by ten princes from each of Saxony, Bavaria, Swabia and Franconia, with the Archbishop of Mainz presiding.
mb_hre_electors_stem_princes_effect = {
	add_to_global_variable_list = { name = hre_elector_list target = title:c_mainz }		# Archbishop of Mainz, archchancellor, who led the election
	add_to_global_variable_list = { name = hre_elector_list target = title:c_cologne }		# Archbishop of Cologne, who crowned the king at Aachen
	add_to_global_variable_list = { name = hre_elector_list target = title:c_trier }		# Archbishop of Trier
	add_to_global_variable_list = { name = hre_elector_list target = title:d_anhalt }		# Duke of Saxony
	add_to_global_variable_list = { name = hre_elector_list target = title:d_oberbayern }	# Duke of Bavaria
	add_to_global_variable_list = { name = hre_elector_list target = title:d_swabia }		# Duke of Swabia
	add_to_global_variable_list = { name = hre_elector_list target = title:d_brabant }		# Duke of Lower Lorraine
	set_global_variable = { name = mb_hre_stem_prince_electors value = yes }
}

# The seven prince-electors: the three Rhenish archbishops, the King of Bohemia, the Count Palatine of the Rhine, the Duke of Saxony and the Margrave of Brandenburg.
# They emerge in the double election of 1198, are listed (without Bohemia) in the Sachsenspiegel around 1230, act as a body in the double election of 1257, and are fixed by the Golden Bull of 1356.
mb_hre_electors_seven_effect = {
	add_to_global_variable_list = { name = hre_elector_list target = title:c_mainz }		# Archbishop of Mainz
	add_to_global_variable_list = { name = hre_elector_list target = title:c_cologne }		# Archbishop of Cologne
	add_to_global_variable_list = { name = hre_elector_list target = title:c_trier }		# Archbishop of Trier
	if = {	# King of Bohemia (royal title from Ottokar I in 1198)
		limit = { exists = title:k_bohemia.holder }
		add_to_global_variable_list = { name = hre_elector_list target = title:k_bohemia }
	}
	else = {
		add_to_global_variable_list = { name = hre_elector_list target = title:d_bohemia }
	}
	if = {	# Count Palatine of the Rhine
		limit = { exists = title:d_pfalz.holder }
		add_to_global_variable_list = { name = hre_elector_list target = title:d_pfalz }
	}
	else = {
		add_to_global_variable_list = { name = hre_elector_list target = title:c_speyer }
	}
	add_to_global_variable_list = { name = hre_elector_list target = title:d_anhalt }		# Duke of Saxony (Wittenberg line, confirmed by the Golden Bull over Lauenburg)
	add_to_global_variable_list = { name = hre_elector_list target = title:d_ostmark }		# Margrave of Brandenburg
}

# Empties the list and the tooltip variables that hre_elector_list_save_effect fills (it only fills empty slots)
mb_hre_electors_clear_effect = {
	if = {
		limit = { has_global_variable_list = hre_elector_list }
		clear_global_variable_list = hre_elector_list
	}
	title:e_hre = {
		remove_variable = hre_elector_1
		remove_variable = hre_elector_2
		remove_variable = hre_elector_3
		remove_variable = hre_elector_4
		remove_variable = hre_elector_5
		remove_variable = hre_elector_6
		remove_variable = hre_elector_7
	}
}
