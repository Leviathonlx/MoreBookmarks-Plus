# Overrides of effects from 01_exp1_historical_artifacts_creation_effect.txt

create_artifact_edmund_head_effect = {
	# Get the character the artifact is being made for.
	$OWNER$ = { save_scope_as = owner }
	# Not really used, but if we don't set the scopes we get errors in the feature selection
	set_artifact_rarity_illustrious = yes

	# Create the artifact
	create_artifact = {	
		name = edmund_head_name
		description = edmund_head_description
		type = pedestal
		visuals = head
		wealth = scope:wealth
		quality = scope:quality
		template = christian_relic_template
		history = {
			type = created
			date = 869.11.20
			recipient = character:163064 #Eadmund the Martyr
			location = province:1520 #Blything
		}
		modifier = edmund_head_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}

	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		set_variable = edmund_head
		add_scaled_artifact_modifier_devotion_effect = yes
		set_variable = {
			name = relic_religion
			value = religion:christianity_religion
		}
		save_scope_as = epic
		add_artifact_history = {
			type = given
			date = 945.1.1
			recipient = character:102 # Eadmund I, who granted Beodricesworth to the community of Saint Edmund
		}
		add_artifact_title_history = {
			target = title:k_england
			date = 955.11.24
		}
	}
}

create_artifact_MB_holy_lance_empire_effect = {

	$OWNER$ = { save_scope_as = owner }
	set_artifact_rarity_illustrious = yes

	# Create the artifact
	if = {
		limit = { has_royal_court = yes }
		create_artifact = {
			name = MB_artifact_holy_lance_empire_name
			description = MB_artifact_holy_lance_empire_desc
			type = pedestal
			template = christian_artifact_template
			visuals = pedestal_longinus
			wealth = scope:wealth
			quality = scope:quality
			history = {
				type = created_before_history
			}
			modifier = artifact_knight_effectiveness_9_modifier
			modifier = artifact_monthly_piety_5_modifier
			modifier = artifact_clergy_opinion_4_modifier
			save_scope_as = newly_created_artifact
			decaying = no
		}
	}
	else = {
		create_artifact = {
			name = MB_artifact_holy_lance_empire_name
			description = MB_artifact_holy_lance_empire_desc
			type = miscellaneous_when_not_court
			template = christian_artifact_template
			visuals = pedestal_longinus
			wealth = scope:wealth
			quality = scope:quality
			history = {
				type = created_before_history
			}
			modifier = artifact_knight_effectiveness_9_modifier
			modifier = artifact_monthly_piety_5_modifier
			modifier = artifact_clergy_opinion_4_modifier
			save_scope_as = newly_created_artifact
			decaying = no
		}
	}

	scope:newly_created_artifact = {
		set_variable = { name = historical_unique_artifact value = yes }
		# Flag this as something that can be reforged for free, if relevant.
		if = {
			limit = { scope:owner = { has_royal_court = no } }
			set_variable = { name = dlc_potential_court_artifact value = yes }
		}
		set_variable = {
			name = relic_religion
			value = religion:christianity_religion
		}
		if = {
			limit = { scope:owner = { has_title = title:e_hre } }
			set_variable = { name = artifact_succession_title value = title:e_hre }
		}
		else_if = {
			limit = { scope:owner = { has_title = title:e_germany } }
			set_variable = { name = artifact_succession_title value = title:e_germany }
		}
		add_artifact_history = {
			type = given
			date = 926.11.1
			recipient = character:1280 #Heinrich I
			location = province:2731 #Worms
		}
		add_artifact_title_history = {
			target = title:e_hre
			date = 962.2.2
		}
		save_scope_as = epic
	}
}
