### Global effect that affects every Rite in the game, adding missing doctrines for newly-created slots
mz_fix_religions_effect = {
	if = {
		limit = { # Double-check to make certain this has not already happened
			NOT = { exists = global_var:mz_flag_fix_religion_decision }	
		}
		set_global_variable = { # Add global variable to prevent this from happening more than once
			name = mz_flag_fix_religion_decision
			value = yes
		}
		# Begin process of replacing doctrines and tenets
		every_religion_global = {
			every_faith = {
				every_faith_rite = {
					### Syncretism Doctrines
					#if = { # If we have a Syncretic Tenet, become Syncretized
					#	limit = { mz_rite_has_syncretic_tenet = yes }
					#	change_rite_doctrine = mz_doctrine_is_syncretized
					#}
					#else = { change_rite_doctrine = mz_doctrine_no_syncretism } # Otherwise get One Faith

					## Nudity Doctrines
					if = { # Faiths that have Natural Primitivism have Full Nudity doctrine
						limit = { 
							rite_has_tenet = tenet_natural_primitivism
							NOT = { rite_has_doctrine = mz_doctrine_full_nudity }
						}
						change_rite_doctrine = mz_doctrine_full_nudity
					}
					if = { # All other faiths that lack Full Nudity or Naked Priests have No Nudity doctrine
						limit = { 
							NOR = { 
								rite_has_doctrine = mz_doctrine_full_nudity
								rite_has_doctrine = mz_doctrine_no_nudity
								rite_has_doctrine = special_doctrine_naked_priests
							}
						}
						change_rite_doctrine = mz_doctrine_no_nudity
					}
					
					## Marital Relations
					if = {
						limit = { 
							rite_has_tenet = tenet_polyamory
							NOT = { rite_has_doctrine = mz_doctrine_polyamory }
						}
						change_rite_doctrine = mz_doctrine_polyamory
					}
					else_if = {
						limit = {
							rite_has_parameter = adultery_female_accepted
							rite_has_parameter = adultery_male_accepted
							NOT = { rite_has_doctrine = mz_doctrine_polyamory }
						}
						change_rite_doctrine = mz_doctrine_polyamory
					}
					else = { change_rite_doctrine = mz_doctrine_matrimony }
				}
			}
		}
	}
}

### Discover any doctrine that has an associated tenet, as long as you know the tenet, Character Scope
mz_learn_doctrines_from_tenets = {
	if = {
		limit = { is_playable_character = yes } # This only works for playable characters anyway.
		# Syncretism
		if = {
			limit = { 
				NOT = {
					knows_doctrine = mz_doctrine_no_syncretism
				}
			}
			add_known_doctrine = mz_doctrine_no_syncretism
		}
		if = {
			limit = { 
				NOT = {
					knows_doctrine = mz_doctrine_is_syncretized
				}
				OR = {
					knows_tenet = tenet_sinitic_syncretism
					knows_tenet = tenet_eastern_syncretism
					knows_tenet = tenet_unreformed_syncretism
					knows_tenet = tenet_christian_syncretism
					knows_tenet = tenet_islamic_syncretism
					knows_tenet = tenet_jewish_syncretism
				}				
			}
			add_known_doctrine = mz_doctrine_is_syncretized
		}
		if = {
			limit = { 
				knows_tenet = tenet_sinitic_syncretism
				NOT = {
					knows_doctrine = mz_doctrine_sinitic_syncretism
