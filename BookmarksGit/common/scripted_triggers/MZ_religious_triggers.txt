# Use Rite Scope
mz_rite_has_syncretic_doctrine = {
	OR = {
		rite_has_doctrine = mz_doctrine_sinitic_syncretism
		rite_has_doctrine = mz_doctrine_eastern_syncretism
		rite_has_doctrine = mz_doctrine_unreformed_syncretism
		rite_has_doctrine = mz_doctrine_christian_syncretism
		rite_has_doctrine = mz_doctrine_jewish_syncretism
		rite_has_doctrine = mz_doctrine_islamic_syncretism
	}
}

mz_rite_has_syncretic_tenet = {
	OR = {
		rite_has_tenet = tenet_sinitic_syncretism
		rite_has_tenet = tenet_eastern_syncretism
		rite_has_tenet = tenet_unreformed_syncretism
		rite_has_tenet = tenet_christian_syncretism
		rite_has_tenet = tenet_jewish_syncretism
		rite_has_tenet = tenet_islamic_syncretism
	}
}

mz_rite_has_syncretic_doctrine_selection = {
	OR = {
		doctrine:mz_doctrine_sinitic_syncretism = { is_in_list = selected_doctrines }
		doctrine:mz_doctrine_eastern_syncretism = { is_in_list = selected_doctrines }
		doctrine:mz_doctrine_unreformed_syncretism = { is_in_list = selected_doctrines }
		doctrine:mz_doctrine_christian_syncretism = { is_in_list = selected_doctrines }
		doctrine:mz_doctrine_jewish_syncretism = { is_in_list = selected_doctrines }
		doctrine:mz_doctrine_islamic_syncretism = { is_in_list = selected_doctrines }
	}
}

mz_requires_syncretic_tenets_trigger = {
	custom_description = {
		text = mz_requires_tenet_syncretism_trigger
		OR = {
			tenet:tenet_eastern_syncretism = { is_in_list = selected_doctrines }
			tenet:tenet_unreformed_syncretism = { is_in_list = selected_doctrines }
			tenet:tenet_christian_syncretism = { is_in_list = selected_doctrines }
			tenet:tenet_islamic_syncretism = { is_in_list = selected_doctrines }
			tenet:tenet_jewish_syncretism = { is_in_list = selected_doctrines }
			tenet:tenet_sinitic_syncretism = { is_in_list = selected_doctrines }
		}
	}	
}
