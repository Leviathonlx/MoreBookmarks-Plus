# Faith limits for preset saints (scope: faith)

is_non_chalcedonianism_christian_faith = {
	OR = {
		this = faith:nestorian
		this = faith:miaphysitism
		this = faith:armenian_apostolic
		this = faith:jacobite
		this = faith:julianists
	}
}

# Faiths that rejected the veneration of saints and relics; they receive no preset saints
MB_rejects_saint_cult_faith_trigger = {
	OR = {
		this = faith:cathar
		this = faith:bogomilist
		this = faith:paulician
		this = faith:tondrakian
		this = faith:waldensian
		this = faith:lollard
		this = faith:amalricians
		this = faith:free_spirit
		this = faith:adamites
	}
}

# Apostles and the martyrs before Nicaea (325), honoured by every church that kept the cult of saints
MB_universal_saint_faith_trigger = {
	MB_rejects_saint_cult_faith_trigger = no
}

# Nicene Fathers and saints remembered for opposing Arius
MB_nicene_saint_faith_trigger = {
	MB_universal_saint_faith_trigger = yes
	NOT = { this = faith:arianism }
}

# Saints after Ephesus (431), rejected by the Church of the East
MB_ephesine_saint_faith_trigger = {
	MB_nicene_saint_faith_trigger = yes
	NOT = { this = faith:nestorian }
}

# Saints after Chalcedon (451) honoured by the churches that accepted the council
MB_chalcedonian_saint_faith_trigger = {
	MB_nicene_saint_faith_trigger = yes
	is_non_chalcedonianism_christian_faith = no
}
