trait_is_criminal_in_rite_trigger = {
	OR = {
		## ME Addition - Alcohol Consumption ##
		AND = {
			scope:criminal_trait_check = flag:drunkard
			$RITE$ = { rite_has_parameter = me_intoxication_illegal }
		}								 
		#KINSLAYER
		AND = {
			OR = {
				$TRAIT$ = trait:kinslayer_3
				$TRAIT$ = trait:kinslayer_2
				$TRAIT$ = trait:kinslayer_1
			}
			$RITE$ = { rite_has_parameter = kinslaying_any_dynasty_member_crime }
		}
		AND = {
			OR = {
				$TRAIT$ = trait:kinslayer_3
				$TRAIT$ = trait:kinslayer_2
			}
			$RITE$ = { rite_has_parameter = kinslaying_extended_family_crime }
		}
		AND = {
			$TRAIT$ = trait:kinslayer_3
			$RITE$ = { rite_has_parameter = kinslaying_close_kin_crime }
		}
		#WITCH
		AND = {
			$TRAIT$ = trait:witch
			$RITE$ = { rite_has_parameter = witchcraft_illegal }
		}
		#CANNIBAL
		AND = {
			$TRAIT$ = trait:cannibal
			NOT = {
				$RITE$ = { rite_has_parameter = cannibalism_legal }
			}
		}
		#SODOMITE
		AND = {
			$TRAIT$ = trait:sodomite
			$RITE$ = { rite_has_parameter = homosexuality_illegal }
		}
		#DEVIANT
		AND = {
			$TRAIT$ = trait:deviant
			$RITE$ = { rite_has_parameter = deviancy_illegal }
		}
		#INCEST
		AND = {
			$TRAIT$ = trait:incestuous
			NOT = {
				$RITE$ = { rite_has_parameter = allows_unrestricted_marriage }
			}
		}
		#ADULTERER/FORNICATOR
		AND = {
			OR = {
				$TRAIT$ = trait:adulterer
				$TRAIT$ = trait:fornicator
			}
			OR = {
				AND = {
					$RITE$ = { rite_has_parameter = adultery_female_crime }
					$GENDER_CHARACTER$ = { is_female = yes }
				}
				AND = {
					$RITE$ = { rite_has_parameter = adultery_male_crime }
					$GENDER_CHARACTER$ = { is_male = yes }
				}
			}
		}
		# Devoted is criminal when monasticism is forbidden
		AND = {
			$TRAIT$ = trait:devoted
			$RITE$ = { rite_has_parameter = devoted_trait_criminal }
		}
	}
}
