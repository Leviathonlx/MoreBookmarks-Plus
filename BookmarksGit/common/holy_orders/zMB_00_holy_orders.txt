#Horsemen of Svantevit - the three hundred mounted warriors of the temple of Svantevit at Arkona
ho_arkona_horsemen = {
	type = military
	government = holy_order_government
	name = "holy_order_arkona_horsemen"
	title = d_arkona_horsemen
	coat_of_arms = "ho_arkona_horsemen"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			scope:province.barony ?= title:b_stralsund
		}
		trigger_else = {
			scope:founder.faith = faith:slavic_pagan
		}
	}

	weight = {
		value = 10
		if = {
			limit = {
				scope:province.county ?= title:c_rugen
			}
			add = 100
		}
		if = {
			limit = {
				scope:founder.culture ?= {
					has_cultural_pillar = heritage_west_slavic
				}
			}
			add = 20
		}
		#Trying to put here on game start
		if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
				scope:province.barony ?= title:b_stralsund
			}
			add = 10000
		}
		if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
				NOT = { scope:province.barony ?= title:b_stralsund }
			}
			multiply = 0
		}
	}
	effect = {
		scope:title = {
			set_variable = {
				name = order_heritage
				value = culture_pillar:heritage_west_slavic
			}
		}
	}

	men_at_arms = { light_horsemen }
}

#Jomsvikings - the sworn brotherhood of Jomsborg at the mouth of the Oder
ho_jomsvikings = {
	type = military
	government = holy_order_government
	name = "holy_order_jomsvikings"
	title = d_jomsvikings
	coat_of_arms = "ho_jomsvikings"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			scope:province.barony ?= title:b_kolobrzeg
		}
		trigger_else = {
			scope:founder.faith = faith:norse_pagan
		}
	}

	weight = {
		value = 10
		if = {
			limit = {
				scope:province.county ?= title:c_kolobrzeg
			}
			add = 100
		}
		if = {
			limit = {
				scope:province = {
					county = {
						is_coastal_county = yes
					}
				}
			}
			multiply = 5
		}
		#Trying to put here on game start
		if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
				scope:province.barony ?= title:b_kolobrzeg
			}
			add = 10000
		}
		if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
				NOT = { scope:province.barony ?= title:b_kolobrzeg }
			}
			multiply = 0
		}
	}
	effect = {
		scope:title = {
			set_variable = {
				name = order_heritage
				value = culture_pillar:heritage_north_germanic
			}
		}
	}

	men_at_arms = { jomsviking_pirates }
}
