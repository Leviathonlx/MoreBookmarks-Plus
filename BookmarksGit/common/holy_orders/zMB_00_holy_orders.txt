#Horsemen of Svantevit - the three hundred mounted warriors of the temple of Svantevit at Arkona
ho_arkona_horsemen = {
	type = military
	government = holy_order_government
	name = "holy_order_arkona_horsemen"
	title = d_arkona_horsemen
	coat_of_arms = "ho_arkona_horsemen"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			scope:province.barony ?= title:b_stralsund
		}
		trigger_else = {
			scope:founder.faith = faith:slavic_pagan
		}
	}

	weight = {
		value = 10
		if = {
			limit = {
				scope:province.county ?= title:c_rugen
			}
			add = 100
		}
		if = {
			limit = {
				scope:founder.culture ?= {
					has_cultural_pillar = heritage_west_slavic
				}
			}
			add = 20
		}
		#Trying to put here on game start
		if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
				scope:province.barony ?= title:b_stralsund
			}
			add = 10000
		}
		if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
				NOT = { scope:province.barony ?= title:b_stralsund }
			}
			multiply = 0
		}
	}
	effect = {
		scope:title = {
			set_variable = {
				name = order_heritage
				value = culture_pillar:heritage_west_slavic
			}
		}
	}

	men_at_arms = { light_horsemen }
}

#Mattamayuras - the Shaiva Siddhanta ascetics of the Mattamayura lineage, with mathas at Ranod, Kadwaha, Terahi and Surwaya and, under the Kalachuris, at Chandrehe, Gurgi and Bilhari
ho_mattamayura = {

	type = monastic
	government = monastic_holy_order_government
	name = "holy_order_mattamayura"
	title = d_mattamayura
	coat_of_arms = "ho_mattamayura"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
			}
			scope:founder = { has_variable = mattamayura_founder }
			OR = {
				scope:province.barony ?= title:b_bahoriband
				scope:province.barony ?= title:b_chandrehi
			}
		}
		trigger_else = {
			scope:founder.rite = rite:mattamayura
			scope:province = {
				OR = {
					county = title:c_gurgi
					county = title:c_tripuri
					county = title:c_chanderi
					county = title:c_candhoba
					county = title:c_bandhugadha
					county = title:c_damoh
				}
			}
		}
	}

	weight = {
		value = 10

		#The Mattamayura mathas of Chedi and of the country of King Avanti
		if = {
			limit = {
				OR = {
					scope:province.county ?= title:c_gurgi
					scope:province.county ?= title:c_candhoba
				}
			}
			multiply = 10
		}
		#Trying to put them here on game start
		if = {
			limit = {
				scope:founder = {
					has_variable = monastic_game_start_founder
					has_variable = mattamayura_founder
				}
				OR = {
					scope:province.barony ?= title:b_bahoriband
					scope:province.barony ?= title:b_chandrehi
				}
			}
			add = 10000
		}
	}

	effect = {
		if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
			}
			scope:new_holy_order = { set_order_tenet = tenet:tenet_esotericism }
		}
	}

}

	name = "holy_order_jimon"
	title = d_jimon
	coat_of_arms = "ho_jimon"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
			}
			scope:founder = { has_variable = jimon_founder }
			scope:province.barony ?= title:b_azai
		}
		trigger_else = {
			scope:founder.rite = rite:pundarika
			scope:province = {
				OR = {
					county = title:c_mino
					county = title:c_hida
					county = title:c_shima
					county = title:c_omi
				}
			}
		}
	}

	weight = {
		value = 10

		#Onjo-ji at Otsu
		if = {
			limit = {
				OR = {
					scope:province.county ?= title:c_omi
				}
			}
			multiply = 10
		}
		#Trying to put them here on game start
		if = {
			limit = {
				scope:founder = {
					has_variable = monastic_game_start_founder
					has_variable = jimon_founder
				}
				scope:province.barony ?= title:b_azai
			}
			add = 10000
		}
	}

	effect = {
		if = {
			limit = {
				scope:founder = { has_variable = monastic_game_start_founder }
			}
			scope:new_holy_order = { set_order_tenet = tenet:tenet_esotericism }
		}
	}

}
