#Knights Templar
ho_christian_knights_templar = {
	
	type = military
	government = holy_order_government
	title = d_knights_templar
	name = "holy_order_knights_templar"
	coat_of_arms = "ho_knights_templar"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			OR = {
				scope:province.barony ?= title:b_towerdavid
				scope:province.barony ?= title:b_acre
				scope:province.barony ?= title:b_kantara
			}
		}
		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
			}
		}
	}

	weight = {
		value = 10
		#Should be founded in jerusalem
		if = {
			limit = {
				scope:province.county ?= title:c_jerusalem
			}
			add = 1000
		}
		if = {
			limit = {
				scope:province = {
					geographical_region = world_middle_east_jerusalem
				}
			}
			add = 100
		}
		#More likely if first holy order
		if = {
			limit = {
				scope:founder.faith = {
					NOT = {
						any_faith_holy_order = {
							leader ?= {
								government_has_flag = government_is_military_holy_order
							}
						}
					}
				}
			}
			add = 100
		}
		
		#Less likely outside age of crusades
		if = {
			limit = {
				NOT = {
					scope:founder.religion = { exists = var:variable_ghw_unlocked }
				}
			}
			multiply = 0.1
		}
		#Trying to put them here on game start
		if = {
			limit = {
				years_from_game_start < 1
				OR = {
					scope:province.barony ?= title:b_towerdavid
					scope:province.barony ?= title:b_acre
					scope:province.barony ?= title:b_kantara
				}
			}
			add = 1000
		}
	}

	men_at_arms = { order_knights }

}

#Knights Hospitaller
ho_christian_knights_hospitaller = {
	
	type = military
	government = holy_order_government
	title = d_knights_hospitaller
	name = "holy_order_knights_hospitaller"
	coat_of_arms = "ho_knights_hospitaller"

	trigger = {
		trigger_if = {
			limit = {
				scope:founder = { has_variable = military_order_game_start_founder }
			}
			OR = {
				scope:province.barony ?= title:b_bethlehem
				scope:province.barony ?= title:b_tyre
				scope:province.barony ?= title:b_paphos
				scope:province.barony ?= title:b_lindos
			}
		}
		trigger_else = {
			scope:founder = {
				faith = {
					has_doctrine_parameter = is_western_christian_faith
				}
				# founder is a hospital-building type
				ai_compassion >= 0
			}
		}
	}

	weight = {
		value = 10
		#Should be founded in jerusalem
		if = {
			limit = {
				scope:province.county ?= title:c_jerusalem
			}
			add = 1000
		}
		if = {
			limit = {
				scope:province = {
					geographical_region = world_middle_east_jerusalem
				}
			}
			add = 50
		}
		#More likely if second holy order
		if = {
			limit = {
				scope:founder.faith = {
					any_faith_holy_order = {
						count < 2
						leader ?= {
							government_has_flag = government_is_military_holy_order
						}
					}
				}
			}
			add = 50
		}
		#More likely if founder is a hospital-building type
		if = {
			limit = {
				scope:founder = {
					ai_compassion >= high_positive_ai_value
				}
			}
			add = 100
		}
		#Less likely outside age of crusades
		if = {
			limit = {
				NOT = {
					scope:founder.religion = { exists = var:variable_ghw_unlocked }
				}
			}
			multiply = 0.1
		}
		#Trying to put them here on game start
		if = {
			limit = {
				years_from_game_start < 1
				OR = {
					scope:province.barony ?= title:b_bethgibelin
					scope:province.barony ?= title:b_amioun
					scope:province.barony ?= title:b_paphos
					scope:province.barony ?= title:b_lindos
					}
				}
			add = 10000
		}
	}

	men_at_arms = { order_knights }

	effect = {
		scope:leader = {
			add_trait = lifestyle_physician
			add_trait_xp = {
				trait = lifestyle_physician
				value = { 25 50 }
			}
		}
	}
}

		}

		}

			scope:founder = {

				}
			}
			#Pastoral-like
			scope:province.county = {
				any_county_province = {
					has_holding = no
				}
				OR = {
					scope:province.county = {
						county_control < 50
					}
					NOT = {
						scope:province.county = {
							save_temporary_scope_as = county_temp
							any_neighboring_county = {
								development_level < scope:county_temp.development_level
							}
						}
					}
				}
			}
		}
	}

	weight = {
		value = 1

		#More likely if the founder's rite values seclusion
		if = {
			limit = {
				scope:founder.rite ?= {
					OR = {
						rite_has_tenet = tenet_pastoral_isolation
						has_tenet_status = {
							tenet = tenet_pastoral_isolation
							status = permitted
						}
					}
				}
			}
			multiply = 2
		}

		#More likely if related to Benedictines

			limit = {
				scope:founder = {
					has_variable = benedictine
				}
			}
			add = 10
		}
		#Trying to put them here on game start

			limit = {

					has_variable = monastic_game_start_founder

				}
