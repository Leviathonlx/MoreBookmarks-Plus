# MB - the whole Aksumite line (aksum, ethiopian, tigray, argobba)
tradition_amharic_highlanders = { #Ethiopian Unique Tradition, replaces hill dwellers, Unlocks MAA innovation & Castration
	category = realm

	layers = {
		0 = steward
		1 = mena
		4 = mountain.dds
	}

	is_shown = {
		NOT = { has_cultural_tradition = tradition_hill_dwellers }
		culture_is_or_has_ancestor_trigger = { CULTURE = culture:aksum }
	}
	can_pick = {
		custom_description = {
			text = culture_in_hills_desc
			any_culture_county = {
				county_has_province_with_terrain = {
					terrain = hills
				}
			}
		}
	}
	
	parameters = {
		unlock_sarawit_innovation = yes
		can_castrate_prisoners = yes
		hills_villages_building_slot = yes
		hills_villages_defender_bonuses = yes
		terrace_gardens_hills = yes
	}
	county_modifier = {
		hills_prosperity_growth_factor = 0.2
	}
	character_modifier = {
		hills_advantage = 5
		hills_travel_danger = hills_high_danger_reduction
		mountains_travel_danger = mountains_low_danger_reduction
		desert_mountains_travel_danger = desert_mountains_low_danger_reduction
		trade_good_millet_buy_price_mult = -0.015
	}

	cost = {
		prestige = {
			add = {
				value = tradition_base_cost
				desc = BASE
				format = "BASE_VALUE_FORMAT"
			}
			if = {
				limit = {
					NOT = {
						culture_pillar:ethos_spiritual = { is_in_list = traits }
					}
				}
				add = {
					value = tradition_incompatible_ethos_penalty
					desc = not_spiritual_desc
				}
			}
			if = {
				limit = {
					NOT = {
						any_culture_county = {
							percent >= 0.3
							county_has_province_with_terrain = {
								terrain = hills
							}
						}
					}
				}
				add = {
					value = tradition_unfulfilled_criteria_penalty
					desc = hill_percentage_desc
				}
			}

			if = {
				limit = {
					culture_tradition_reduction_trigger = { TRADITION = tradition_amharic_highlanders }
				}
				multiply = {
					value = 0.5
					desc = inspired_by_culture_desc
				}
			}
			
			multiply = tradition_replacement_cost_if_relevant
		}
	}
	
	ai_will_do = {
		value = 400
		if = {
			limit = {
				NOT = {
					scope:character = {
						any_sub_realm_county = {
							percent >= 0.2
							culture = scope:character.culture
							county_has_province_with_terrain = {
								terrain = hills
							}
						}
					}
				}
			}
			multiply = 0
		}
	}
}

	category = regional

	layers = {
		0 = martial
		1 = mena
		4 = soldiers.dds
	}
	
	#Can pick this if you have arabic heritage.
	is_shown = { #Replaces Warriors of the Dry
		OR = { has_cultural_pillar = heritage_arabic has_cultural_pillar = heritage_south_arabian }
	}
	can_pick = {
		culture_not_pacifistic_trigger = yes
		trigger_if = {
			limit = {
				OR = {
					NOT = { exists = scope:replacing }
					NOT = { scope:replacing = culture_tradition:tradition_warriors_of_the_dry }
				}
			}
			NOT = { has_cultural_tradition = tradition_warriors_of_the_dry } #To prevent stacking after hybridization
