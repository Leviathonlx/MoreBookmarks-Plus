egyptian_cult_isis_doctrine = {		# Isis
	doctrine_group_type = doctrine_egyptian_cult
	index = 0
	divergence = 10
	
	piety_cost = {
		value = faith_doctrine_cost_mid
		if = {
			limit = { rite_has_doctrine = egyptian_cult_isis_doctrine }

		}
		else = {
			value = faith_tenet_cost_low
		}
	}
	parameters = {
		TFE_egyptian_cult_isis_extra_bonuses = yes

}
