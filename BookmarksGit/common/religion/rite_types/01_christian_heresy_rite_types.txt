# These rites are spawned exclusively via the heresy outbreak system (Component 3).
# They have no parent faith and no bookmark blocks,
# so they will not be generated from history; create_rite_from_type is the only entry point.
# Pre-declaration: founder is in the parent Christian faith (catholic / orthodox / christian_faith),
# so the rite materialises as a divergent rite of that faith.

cathar_rite = {
    name = cathar_rite
    desc = cathar_rite_desc
    faith = cathar
    convert = no

    color = { 0.5 0.5 0.9 }
    icon = christianity_cathar

	tenets = {
		tenet_consolamentum # The one central rite, which removed sin and made a believer one of the Perfect; many received it near death
		tenet_pacifism # Killing was abhorrent to them: they condemned war and capital punishment
		tenet_gnosticism # A spiritual world of the good God set against a material world made by the evil power
	}

	doctrines = {
		doctrine_no_head # Bishops at Albi (c. 1165), and at Toulouse, Carcassonne and Agen after the council of Saint-Felix (1167), but no single head
		doctrine_clerical_gender_either # Women could become Perfects and give the consolamentum
		doctrine_sacraments_absent # They rejected the Eucharist and baptism in water
		doctrine_no_anointment
		special_doctrine_is_western_christian_faith
		doctrine_me_intoxication_accepted
		doctrine_me_ablution_handwashing
	}

	localization = {
		PriestMale = cathar_priest_male
		PriestMalePlural = cathar_priest_male_plural
		PriestFemale = cathar_priest_female
		PriestFemalePlural = cathar_priest_female_plural
		PriestNeuter = cathar_priest_neuter
		PriestNeuterPlural = cathar_priest_neuter_plural
	}
}
