# On actions for when the player character changes, triggered automatically by the game

# For mods that want to intercept all cases where a player changes character from one to another
# Root is the new player character
# scope:previous_player_character is the old player character
on_player_character_change = {
	effect = {
		if = {
			limit = { exists = puppeteer }
			clear_puppet_status = yes
			remove_puppet_modifiers_effect = yes
		}
		pam_offer_great_schism_stance_to_new_player_effect = yes
	}
}
