natural_primitivism_grant_fulfillment_effect = {
	if = {
		limit = {
			is_playable_character = yes
			natural_primitivism_fulfillment_trigger = yes
		}
		change_spiritual_fulfillment = major_spiritual_fulfillment_value
	}
}

natural_primitivism_apply_court_naturism_to_character_effect = {
	if = {
		limit = {
			NOT = { has_character_modifier = natural_primitivism_court_naturism_modifier }
		}
		add_character_modifier = { modifier = natural_primitivism_court_naturism_modifier }
		add_stress = major_stress_loss
		natural_primitivism_grant_fulfillment_effect = yes
	}
	hidden_effect = {
		remove_character_flag = natural_primitivism_rejected_court_naturism
	}
}

natural_primitivism_promote_court_naturism_effect = {
	hidden_effect = {
		save_scope_as = naturism_liege
		add_character_flag = natural_primitivism_promoted_naturism_active
	}
	natural_primitivism_apply_court_naturism_to_character_effect = yes
	if = {
		limit = { exists = capital_county }
		capital_county = {
			hidden_effect = {
				change_county_control = minor_county_control_loss
			}
			add_county_modifier = {
				modifier = natural_primitivism_unbound_court_county_modifier
				years = 5
			}
		}
	}
	hidden_effect = {
		every_courtier = {
			limit = {
				is_available_adult = yes
				has_character_flag = natural_primitivism_rejected_court_naturism
			}
			remove_character_flag = natural_primitivism_rejected_court_naturism
		}
	}
	every_courtier = {
		limit = {
			is_available_adult = yes
			NOT = { has_character_modifier = natural_primitivism_court_naturism_modifier }
		}
		custom = natural_primitivism.every_adult_courtier
		natural_primitivism_apply_court_naturism_to_character_effect = yes
	}
}

natural_primitivism_renounce_court_naturism_effect = {
	hidden_effect = {
		remove_character_flag = natural_primitivism_promoted_naturism_active
	}
	remove_character_modifier = natural_primitivism_court_naturism_modifier
	add_character_modifier = {
		modifier = natural_primitivism_court_naturism_self_conscious_modifier
		years = 5
	}
	if = {
		limit = { exists = capital_county }
		capital_county = {
			remove_county_modifier = natural_primitivism_unbound_court_county_modifier
		}
	}
	every_courtier = {
		limit = {
			is_adult = yes
			has_character_modifier = natural_primitivism_court_naturism_modifier
		}
		custom = natural_primitivism.every_adult_courtier
		remove_character_modifier = natural_primitivism_court_naturism_modifier
	}
	hidden_effect = {
		every_courtier = {
			limit = {
				is_adult = yes
				has_character_flag = natural_primitivism_rejected_court_naturism
			}
			remove_character_flag = natural_primitivism_rejected_court_naturism
		}
	}
}

natural_primitivism_on_join_active_naturist_court_effect = {
	if = {
		limit = {
			is_available_adult = yes
			exists = scope:new_employer
			scope:new_employer = {
				has_character_flag = natural_primitivism_promoted_naturism_active
			}
			NOT = { has_character_modifier = natural_primitivism_court_naturism_modifier }
		}
		hidden_effect = {
			remove_character_flag = natural_primitivism_rejected_court_naturism
			scope:new_employer = { save_scope_as = naturism_liege }
		}
		if = {
			limit = { is_ai = yes }
			natural_primitivism_apply_court_naturism_to_character_effect = yes
		}
		else = {
			trigger_event = natural_primitivism.0003
		}
	}
}
