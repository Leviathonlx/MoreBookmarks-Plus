###########################################
# Heresy system - historical founders     #
###########################################
# Three flavours of founder per named-historical heresy:
#   primary:    spawn the iconic historical figure (first outbreak only)
#   secondary:  spawn a named secondary historical figure (second outbreak)
#   resurgence: spawn a generic anonymous preacher (third+ outbreak)
# Always-anonymous heresies (Paulician, Iconoclast, Adoptionist, Messalian)
# have a single anonymous template that is reused on every outbreak.

#####################################
# ----- Lollard ------------------- #
#####################################
# Primary: John Wycliffe (~1320-1384, Yorkshire / Lutterworth) https://en.wikipedia.org/wiki/John_Wycliffe
pam_heresy_spawn_lollard_founder_effect = {
	create_character = {
		age = 20
		name = "John"
		dynasty_house = house:house_wycliffe
		gender_female_chance = 0
		random_traits = no
		location = scope:birth_location.title_province
		faith = faith:lollard
		culture = scope:birth_location.title_province.culture
		trait = education_learning_4
		trait = zealous
		trait = brave
		trait = stubborn
		trait = theologian
		trait = devoted
		diplomacy = { min_template_high_skill max_template_high_skill }
		martial = { min_template_low_skill max_template_low_skill }
		stewardship = { min_template_high_skill max_template_high_skill }
		intrigue = { min_template_low_skill max_template_low_skill }
		learning = { min_template_high_skill max_template_high_skill }
		prowess = { min_template_low_skill max_template_low_skill }
		save_scope_as = founder
	}
	scope:founder ?= {
		historical_character_finalization_effect = yes
		set_variable = john_wycliffe
		add_piety = 5000
		add_character_modifier = { modifier = vow_of_poverty_modifier }
	}
}

# These heresies have no historical CK3-window anchor, so each outbreak spawns
# a generic local preacher reviving the cause. Same template every time.

# Paulician: anonymous Asia Minor preacher reviving the dualist Paulician cause
# (Sergius-Tychicus, Karbeas, Chrysocheir all dead before any CK3 bookmark).
pam_heresy_spawn_paulician_founder_effect = {
	create_character = {
		age = 30
		dynasty = none
		gender_female_chance = 0
		random_traits = no
		location = scope:birth_location.title_province
		faith = faith:paulician
		culture = scope:birth_location.title_province.culture
		trait = education_learning_3
		trait = brave
		trait = zealous
		trait = theologian
		trait = stubborn
		diplomacy = { min_template_decent_skill max_template_decent_skill }
		martial = { min_template_decent_skill max_template_decent_skill }
		stewardship = { min_template_low_skill max_template_low_skill }
		intrigue = { min_template_low_skill max_template_low_skill }
		learning = { min_template_high_skill max_template_high_skill }
		prowess = { min_template_decent_skill max_template_decent_skill }
		save_scope_as = founder
	}
}
