######################
### PASSIVE RITE LEARNING
######################
# Yearly passive discovery of tenets, doctrines, and whole rites through
# characters around the learner. Called from the yearly_playable_pulse.

passive_rite_learning_pulse_effect = {
	if = {
		limit = {
			is_available_adult = yes
		}
		passive_rite_learning_pick_source_effect = yes
		if = {
			limit = { exists = scope:passive_learn_source }
			passive_rite_learning_apply_from_source_effect = yes
		}
	}
}

passive_rite_learning_pick_source_effect = {
	if = {
		limit = {
			any_relation = {
				type = soulmate
				is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
			}
		}
		random_relation = {
			type = soulmate
			limit = { is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root } }
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 1 }
	}
	else_if = {
		limit = {
			any_relation = {
				type = best_friend
				is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
			}
		}
		random_relation = {
			type = best_friend
			limit = { is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root } }
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 1 }
	}
	else_if = {
		limit = {
			any_spouse = { is_apex_partner_passive_learning_source_trigger = yes }
		}
		random_spouse = {
			limit = { is_apex_partner_passive_learning_source_trigger = yes }
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 1 }
	}
	else_if = {
		limit = {
			any_spouse = { is_close_partner_passive_learning_source_trigger = yes }
		}
		random_spouse = {
			limit = { is_close_partner_passive_learning_source_trigger = yes }
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 2 }
	}
	else_if = {
		limit = {
			any_relation = {
				type = friend
				is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
			}
		}
		random_relation = {
			type = friend
			limit = { is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root } }
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 2 }
	}
	else_if = {
		limit = {
			any_relation = {
				type = lover
				is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
			}
		}
		random_relation = {
			type = lover
			limit = { is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root } }
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 2 }
	}
	else_if = {
		limit = {
			is_ai = no # Avoid AI's iterating large lists of characters, like courtiers
			any_courtier = {
				is_close_or_extended_family_of = root
				is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
				location ?= root.location
			}
		}
		random_courtier = {
			limit = {
				is_close_or_extended_family_of = root
				is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
				location ?= root.location
			}
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 2 }
	}
	else_if = {
		limit = {
			any_concubine = { is_close_partner_passive_learning_source_trigger = yes }
		}
		random_concubine = {
			limit = { is_close_partner_passive_learning_source_trigger = yes }
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 3 }
	}
	else_if = {
		limit = {
			any_councillor = { is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root } }
		}
		random_councillor = {
			limit = { is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root } }
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 3 }
	}
	else_if = {
		limit = {
			is_ai = no # Avoid AI's iterating large lists of characters, like courtiers
			any_courtier = {
				is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
				location ?= root.location
			}
		}
		random_courtier = {
			limit = {
				is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
				location ?= root.location
			}
			save_scope_as = passive_learn_source
		}
		save_scope_value_as = { name = passive_learn_source_tier value = 3 }
	}
}

passive_rite_learning_apply_from_source_effect = {
	random_list = {
		10 = {
			trigger = {
				scope:passive_learn_source_tier <= 2
				NOT = {
					knows_rite_level = {
						target = scope:passive_learn_source.rite
						value >= 100
					}
				}
			}
			modifier = {
				add = 990
				scope:passive_learn_source_tier = 1
			}
			add_known_rite = scope:passive_learn_source.rite
			send_interface_message = {
				type = msg_passive_learned_rite
				desc = msg_passive_learned_rite_desc
				left_icon = scope:passive_learn_source
				right_icon = scope:passive_learn_source.rite
			}
		}
		45 = {
			trigger = {
				any_character_tenet = {
					knowledge_filter = unknown
					passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:passive_learn_source.rite }
					count >= 1
				}
			}
			passive_rite_learning_pick_and_apply_tenet_effect = yes
		}
		45 = {
			trigger = {
				any_character_doctrine = {
					knowledge_filter = unknown
					passive_rite_learning_candidate_doctrine_is_core_trigger = { RITE = scope:passive_learn_source.rite }
					passive_rite_learning_doctrine_is_learnable_trigger = yes
					count >= 1
				}
			}
			passive_rite_learning_pick_and_apply_doctrine_effect = yes
		}
	}
}

######################
### PASSIVE LEARNING FOR THE HEAD OF RITE
######################
# Driven by on_rite_yearly 
# Heads of Rite pick up core tenets and doctrines from rites within a neighbouring faith

rite_yearly_hor_passive_learning_effect = {
	if = {
		limit = {
			head_of_rite ?= { is_available_adult = yes }
		}
		random = {
			chance = passive_rite_learning_hor_yearly_chance

			head_of_rite = { save_scope_as = passive_learn_hor }

			faith = {
				random_neighboring_faith = {
					limit = {
						this != prev
						any_faith_rite = {
							rite_counties >= 1
							save_temporary_scope_as = candidate_rite
							OR = {
								scope:passive_learn_hor = {
									any_character_tenet = {
										knowledge_filter = unknown
										passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:candidate_rite }
										count >= 1
									}
								}
								scope:passive_learn_hor = {
									any_character_doctrine = {
										knowledge_filter = unknown
										passive_rite_learning_candidate_doctrine_is_core_trigger = { RITE = scope:candidate_rite }
										passive_rite_learning_doctrine_is_learnable_trigger = yes
										count >= 1
									}
								}
							}
						}
					}
					save_scope_as = passive_learn_source_faith
				}
			}

			if = {
				limit = { exists = scope:passive_learn_source_faith }
				scope:passive_learn_source_faith = {
					random_faith_rite = {
						limit = {
							rite_counties >= 1
							save_temporary_scope_as = candidate_rite
							OR = {
								scope:passive_learn_hor = {
									any_character_tenet = {
										knowledge_filter = unknown
										passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:candidate_rite }
										count >= 1
									}
								}
								scope:passive_learn_hor = {
									any_character_doctrine = {
										knowledge_filter = unknown
										passive_rite_learning_candidate_doctrine_is_core_trigger = { RITE = scope:candidate_rite }
										passive_rite_learning_doctrine_is_learnable_trigger = yes
										count >= 1
									}
								}
							}
						}
						save_scope_as = passive_learn_source_rite
					}
				}

				if = {
					limit = { exists = scope:passive_learn_source_rite }
					scope:passive_learn_hor = {
						random_list = {
							70 = {
								trigger = {
									any_character_tenet = {
										knowledge_filter = unknown
										passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:passive_learn_source_rite }
										count >= 1
									}
								}
								rite_yearly_hor_pick_and_apply_tenet_effect = yes
							}
							30 = {
								trigger = {
									any_character_doctrine = {
										knowledge_filter = unknown
										passive_rite_learning_candidate_doctrine_is_core_trigger = { RITE = scope:passive_learn_source_rite }
										passive_rite_learning_doctrine_is_learnable_trigger = yes
										count >= 1
									}
								}
								rite_yearly_hor_pick_and_apply_doctrine_effect = yes
							}
						}
					}
				}
			}
		}
	}
}
