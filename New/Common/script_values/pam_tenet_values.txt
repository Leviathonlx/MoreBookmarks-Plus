# Same gold as ransom_interaction_effect
pam_philanthropia_ransom_gold_value = {
	value = 0
	if = {
		limit = { scope:extortionate_gold ?= yes }
		add = scope:prisoner.increased_ransom_cost_value
	}
	else_if = {
		limit = { scope:gold ?= yes }
		add = scope:prisoner.normal_ransom_cost_value
	}
	else_if = {
		limit = {
			OR = {
				scope:extortionate_current_gold ?= yes
				scope:current_gold ?= yes
			}
		}
		add = scope:ransom_saved_gold_value
	}
	min = 0
}
