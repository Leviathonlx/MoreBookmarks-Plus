# ROOT = character whose faith is read; scope:second = the tenet
# Tier boundaries are kept in step with tenet_popularity_gui_* in pam_scripted_guis.txt
pam_tenet_popularity_status = {
	type = character
	random_valid = no

	text = {
		trigger = {
			"faith.tenet_popularity(scope:second)" >= tenet_popularity_very_popular_threshold
		}
		localization_key = pam_tenet_popularity_status_very_popular
	}
	text = {
		trigger = {
			"faith.tenet_popularity(scope:second)" < tenet_popularity_very_popular_threshold
			"faith.tenet_popularity(scope:second)" >= tenet_popularity_popular_threshold
		}
		localization_key = pam_tenet_popularity_status_popular
	}
	text = {
		trigger = {
			"faith.tenet_popularity(scope:second)" < tenet_popularity_popular_threshold
			"faith.tenet_popularity(scope:second)" >= tenet_popularity_neutral_threshold
		}
		localization_key = pam_tenet_popularity_status_neutral
	}
	text = {
		trigger = {
			"faith.tenet_popularity(scope:second)" < tenet_popularity_neutral_threshold
		}
		localization_key = pam_tenet_popularity_status_unpopular
	}
}
