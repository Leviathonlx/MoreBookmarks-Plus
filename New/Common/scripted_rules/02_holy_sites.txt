# Whether an artifact can be enshrined in a specific holy site.
# root            = the artifact being considered for enshrining
# scope:actor     = the character who would perform the enshrinement
# scope:holy_site = the target holy site
# scope:replacing = the artifact being replaced (if any)
can_enshrine_artifact = {
	custom_description = {
		text = holy_site_not_holder_or_top_liege

		scope:holy_site.holy_site_county = {
			OR = {
				holder = scope:actor
				holder.top_liege = scope:actor
				clerical_region_title.holder = scope:actor
			}
		}
	}
	# Controlling another faith's holy site does not make it a shrine of yours
	custom_description = {
		text = holy_site_of_actor_faith

		scope:actor.faith = {
			is_faith_holy_site = scope:holy_site
		}
	}
}

# Whether a character can create a new dynamic holy site.
# root                 = the character creating the holy site
# scope:title          = the barony the holy site would be placed in
# scope:holy_site_type = the holy site type being created
# scope:faith          = the faith the holy site is created for
can_create_dynamic_holy_site = {
	can_manage_faith_holy_sites_trigger = yes

	# This is a command, not a decision, so the variable cooldown is set in on_character_holy_site_created
	custom_description = {
		text = holy_site_creation_on_cooldown_tt
		NOT = { has_variable = holy_site_creation_cooldown }
	}

	scope:faith = {
		faith_can_use_holy_site_kingdom_trigger = { BARONY = scope:title }
	}
}

# Whether an enshrined artifact can be removed from a holy site.
# root            = the enshrined artifact
# scope:holy_site = the holy site it sits in
# scope:actor     = the character who would perform the removal
# scope:replacing = the artifact being replaced (if any)
can_remove_enshrined_artifact = {
	custom_description = {
		text = holy_site_not_holder_or_top_liege

		scope:holy_site.holy_site_county = {
			OR = {
				holder = scope:actor
				holder.top_liege = scope:actor
				clerical_region_title.holder = scope:actor
			}
		}
	}
}
