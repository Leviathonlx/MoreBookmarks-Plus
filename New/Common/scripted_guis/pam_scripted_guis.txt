# Great Schism progress shown in every Christian Church phase until the schism happens or is prevented
pam_great_schism_progress_gui = {
	scope = character

	is_shown = {
		pam_great_schism_progress_can_change_trigger = yes
	}
}

pam_investiture_crisis_chapter_unavailable_gui = {
	scope = character

	is_shown = {
		pam_main_power_faith_conditional_appointment_trigger = no
		situation:the_christian_church ?= {
			OR = {
				situation_current_phase = fragile_unity
				situation_current_phase = reform_phase
			}
		}
	}
}

pam_has_schism_stance = {
	scope = character

	is_shown = {
		has_variable = pam_schism_stance
		NOT = { var:pam_schism_stance = flag:no_schism }
		pam_great_schism_progress_can_change_trigger = yes
	}
}


using_tenet_popularity_gui = {
	scope = faith

	is_valid = {
		has_tenet_popularity_trigger = yes
	}
}

tenet_popularity_gui_very_popular = {
	scope = tenet

	is_valid = {
		"scope:faith.tenet_popularity(root)" >= tenet_popularity_very_popular_threshold
	}
}
tenet_popularity_gui_popular = {
	scope = tenet

	is_valid = {
		"scope:faith.tenet_popularity(root)" < tenet_popularity_very_popular_threshold
		"scope:faith.tenet_popularity(root)" >= tenet_popularity_popular_threshold
	}
}
tenet_popularity_gui_neutral = {
	scope = tenet

	is_valid = {
		"scope:faith.tenet_popularity(root)" < tenet_popularity_popular_threshold
		"scope:faith.tenet_popularity(root)" >= tenet_popularity_neutral_threshold
	}
}
tenet_popularity_gui_unpopular = {
	scope = tenet

	is_valid = {
		"scope:faith.tenet_popularity(root)" < tenet_popularity_neutral_threshold
	}
}

pam_is_papabile = {
	scope = character

	is_shown = {
		is_papabile = yes
	}
}

pam_know_doctrine = {
	scope = character
	saved_scopes = { doctrine }

	is_valid = {
		OR = {
			knows_doctrine = scope:doctrine
			has_perk = prophet_perk
		}
	}
}
