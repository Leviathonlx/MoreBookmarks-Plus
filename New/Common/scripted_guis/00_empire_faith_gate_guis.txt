empire_faith_gate_hof_check = {
	scope = character

	is_shown = {
		empire_faith_gate_enabled_trigger = yes
		government_subject_to_faith_empire_trigger = yes
		faith_allows_empire_trigger = yes
		scope:landed_title = { tier >= tier_empire }
		OR = {
			AND = {
				faith = { has_doctrine = doctrine_spiritual_head }
				exists = faith.religious_head
			}
			AND = {
				faith = { has_doctrine = doctrine_temporal_head }
				exists = faith.religious_head
			}
		}
	}

	is_valid = {
		trigger_if = {
			limit = {
				faith = { has_doctrine = doctrine_spiritual_head }
				exists = faith.religious_head
			}
			faith.religious_head = {
				opinion = {
					target = root
					value > 0
				}
			}
		}
		trigger_else_if = {
			limit = {
				faith = { has_doctrine = doctrine_temporal_head }
				exists = faith.religious_head
			}
			OR = {
				this = faith.religious_head
				faith.religious_head = {
					opinion = {
						target = root
						value > 0
					}
				}
			}
		}
		trigger_else = {
			always = yes
		}
	}
}

empire_faith_gate_vassals_check = {
	scope = character

	is_shown = {
		empire_faith_gate_enabled_trigger = yes
		government_subject_to_faith_empire_trigger = yes
		faith_allows_empire_trigger = yes
		scope:landed_title = { tier >= tier_empire }
		# Under the No-HoF-exempt setting, headless faiths have no approval requirement to display
		trigger_if = {
			limit = { has_game_rule = empire_faith_gate_no_hof_exempt }
			efg_faith_has_hof_trigger = yes
		}
	}

	is_valid = {
		any_powerful_vassal = {
			percent >= 0.5
			opinion = {
				target = root
				value > 0
			}
		}
	}
}
