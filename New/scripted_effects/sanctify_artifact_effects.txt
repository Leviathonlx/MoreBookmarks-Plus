# Effects used by sanctify_artifact_interaction.
# Root scope: the artifact being sanctified.
# PRIEST = the clergyman performing the rite.
#
# Records the sanctification on the artifact itself: a faith-appropriate blessing epithet on its name,
# and a line of description naming who sanctified it, for which faith. The owner also remembers the rite.
sanctify_artifact_flavor_effect = {
	save_scope_as = sanctified_artifact
	$PRIEST$ = { save_scope_as = sanctifying_priest }

	# Artifacts that already carry a holy, unique or epithet-bearing name keep it, and only gain the description line.
	if = {
		limit = {
			is_holy_relic = no
			has_none_of_variables = {
				name = sanctified_artifact_var
				name = blessed_artifact_var
				name = cursed_artifact_var
				name = historical_unique_artifact
			}
		}
		set_variable = sanctified_artifact_var
		set_artifact_name = sanctified_artifact_name
	}

	set_artifact_description = sanctified_artifact_desc

	artifact_owner = {
		create_character_memory = {
			type = sanctified_artifact_memory
			participants = {
				sanctifying_priest = scope:sanctifying_priest
			}
		}
		scope:new_memory = {
			set_variable = {
				name = sanctified_artifact
				value = scope:sanctified_artifact
			}
		}
	}
}
