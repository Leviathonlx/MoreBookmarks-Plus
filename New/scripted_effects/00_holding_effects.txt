holding_pre_feudalize_effect = {
	county = {
		if = {
			limit = { has_county_modifier = mpo_siberian_permafrost_modifier }
			remove_county_modifier = mpo_siberian_permafrost_modifier
		}
		else_if = {
			limit = { has_county_modifier = mpo_siberian_permafrost_modifier_bad }
			remove_county_modifier = mpo_siberian_permafrost_modifier_bad
		}
		if = {
			limit = {
				prev = { holding_is_nomad = yes }
				NOT = { has_variable = was_razed }
				culture != holder.culture
			}
			if = {
				limit = { culture = { has_same_culture_heritage = prev.holder.culture } }
				add_county_modifier = {
					modifier = mpo_feudalize_nomadic_land_modifier
					years = 5
				}
			}
			else = {
				add_county_modifier = {
					modifier = mpo_feudalize_nomadic_land_modifier
					years = 30
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:actor } }
			holder = { save_scope_as = actor }
		}
		tgp_silk_road_feudalize_holding_effect = yes
		if = {
			limit = { holder != scope:actor }
			holder = {
				# Add some opinion to make it worthwhile
				add_opinion = {
					target = scope:actor
					modifier = grateful_opinion
					opinion = 30
				}
			}
		}
	}
}

holding_pre_tribalize_effect = {
	county = {
		if = {
			limit = {
				prev = { holding_is_nomad = yes }
				NOT = { has_variable = was_razed }
				culture != holder.culture
			}
			add_county_modifier = {
				modifier = mpo_feudalize_nomadic_land_modifier
				years = 5
			}
		}
	}
}

clan_convert_capital_effect = {
	capital_province ?= {
		if = {
			limit = {
				NOR = { 
					has_holding_type = temple_citadel_holding
					has_holding_type = castle_holding
				}
			}
			custom_tooltip = realm_capital_will_change_to_castle
			hidden_effect = {
				if = {
					limit = { holding_is_tribal = yes }
					holding_pre_feudalize_effect = yes
				}
				set_holding_type = castle_holding
			}
		}
	}
}
