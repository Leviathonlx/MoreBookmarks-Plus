### 0007 - All counties under the new archdiocese are successfully converted to Slavic Rite, and the story ends

###	0010 - Conflict with Frankish preachers

slavic_rite.1000 = { # The brothers inform you of their journey to Rome to seek the Pope's approval
	type = character_event
	title = slavic_rite.1000.t
	desc = slavic_rite.1000.desc
	window = big_event_window
	theme = christian
	override_background = {
		reference = throne_room_west
	}
	right_portrait = {
		character = root
		camera = camera_event_very_right_activity
		animation = throne_room_conversation_3
	}
	center_portrait = {
		character = scope:greek_preacher
		camera = camera_event_left_crowning_observation
		animation = debating
	}
	left_portrait = {
		trigger = {
			exists = root.cp:councillor_court_chaplain
		}
		character = root.cp:councillor_court_chaplain
		camera = camera_event_center_pointing_right
		animation = personality_compassionate
	}

	trigger = {
		title:d_et_salzburg.holder ?= {
			is_ai = yes 
			rite != root.rite
		}
	}

	immediate = {
		title:d_et_salzburg.holder = {
			save_scope_as = salzburg_archbishop
		}
		title:k_orthodox.holder = {
			save_scope_as = byzantine_hor
		}
		cp:councillor_court_chaplain = {
			save_scope_as = court_chaplain
		}
		create_character = {
			template = priest_character_template
			faith = faith:christian_faith
			culture = culture:greek
			dynasty = none
			location = root.location
			save_scope_as = greek_preacher
		}
		hidden_effect = {
			add_visiting_courtier = scope:greek_preacher
			scope:greek_preacher = {
				set_character_rite = rite:byzantine_rite
			}
		}
	}

	option = { # Good, let's wait for their report!
		name = slavic_rite.1000.a
		custom_tooltip = slavic_rite.1000.a_tt
		add_piety = medium_piety_gain
		remove_courtier_or_guest = scope:greek_preacher
		reverse_add_opinion = {
			target = scope:salzburg_archbishop
			modifier = angry_opinion
			opinion = -20
		}
		ai_chance = {
			base = 1
		}
	}

	option = { # On a second thought, maybe we don't want to piss off the Franks?
		name = slavic_rite.1000.b
		random_owned_story = {
			type = story_slavic_rite
			end_story = yes
		}
		custom_tooltip = slavic_rite.1000.b_tt
		custom_tooltip = slavic_rite.1000.b_tt_2
		remove_courtier_or_guest = scope:greek_preacher
		reverse_add_opinion = {
			target = scope:salzburg_archbishop
			modifier = pleased_opinion
			opinion = 20
		}
		reverse_add_opinion = {
			target = scope:byzantine_hor
			modifier = angry_opinion
			opinion = -50
		}

		ai_chance = {
			base = 0
		}
	}

	after = {

	}
}

slavic_rite.0001 = { # The brothers inform you of their journey to Rome to seek the Pope's approval
	type = character_event
	title = slavic_rite.0001.t
	desc = slavic_rite.0001.desc
	window = big_event_window
	theme = christian
	override_background = {
		reference = throne_room_west
	}
	right_portrait = {
		character = root
		camera = camera_event_center_pointing_very_left

	}
	center_portrait = {
		character = scope:methodius_scope

		animation = admiration
	}
	left_portrait = {
		character = scope:cyril_scope
		camera = camera_event_group_talking_left_left
		animation = personality_coward
	}
	trigger = { # Do not fire if the story was cancelled in slavic_rite.1000 (the event is pre-scheduled in on_setup)
		any_owned_story = {
			type = story_slavic_rite
		}
	}
	immediate = {
		hidden_effect = {
			add_visiting_courtier = character:slavic_rite_cyril_1
			add_visiting_courtier = character:slavic_rite_methodius_1
			character:slavic_rite_cyril_1 = {
				save_scope_as = cyril_scope
			}
			character:slavic_rite_methodius_1 = {
				save_scope_as = methodius_scope
			}
			title:k_papal_state.holder = {
				save_scope_as = pope_scope
			}
			random_owned_story = {
				type = story_slavic_rite
				set_variable = {
					name = slavic_rite_cyril
					value = scope:cyril_scope
				}
				set_variable = {
					name = slavic_rite_methodius
					value = scope:methodius_scope
				}
			}
			if = {
				limit = {
					exists = var:slavic_rite_cyril # Error suppression
					exists = var:slavic_rite_methodius # Error suppression
				}
			}
		}
	}

	option = { # Goodbye
		name = slavic_rite.0001.a
		add_piety = minor_piety_gain

		ai_chance = {
			base = 1
		}
	}

	option = { # Goodbye with a gift to Rome
		name = slavic_rite.0001.b
		trigger = {
			gold >= medium_gold_value
		}
		set_variable = {
			name = moravian_mission_gift_sent
			value = yes
		}
		add_piety = major_piety_gain
		remove_short_term_gold = medium_gold_value
		custom_tooltip = slavic_rite_moravian_mission_gift_sent_tooltip

		stress_and_fulfillment_impact = {
			greedy = major_stress_impact_gain
			generous = major_stress_impact_loss
		}
		ai_chance = {
			base = 1
		}
	}

	after = { 
		scope:cyril_scope = {
			hidden_effect = {
				add_character_modifier = pam_brothers_travel_safety_modifier
			}
			start_travel_plan = {
				destination = scope:pope_scope.capital_province
				on_arrival_event = slavic_rite.0002
				on_arrival_destinations = all_but_last
				return_trip = yes
			}
		}
		scope:methodius_scope = {
			hidden_effect = {
				add_character_modifier = pam_brothers_travel_safety_modifier
			}
			start_travel_plan = {
				destination = scope:pope_scope.capital_province
				return_trip = yes
			}
		}
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = slavic_rite_travel_toast_title
			left_icon = character:slavic_rite_cyril_1
			right_icon = character:slavic_rite_methodius_1
			custom_tooltip = slavic_rite.0001.a_tt
		}
	}
}

slavic_rite.0005 = { # Methodius is back from Rome, and the Slavic Rite is finally established
	type = character_event
	title = slavic_rite.0005.t
	desc = slavic_rite.0005.desc
	window = big_event_window
	theme = christian
	override_background = {
		reference = throne_room_west
	}
	right_portrait = {
		character = root
		camera = camera_event_center_pointing_very_left
		animation = personality_rational
	}
	center_portrait = {
		character = character:slavic_rite_methodius_1
		camera = camera_event_left_to_the_left_pointing_in
		animation = holding_scrolls
	}
	lower_left_portrait = {
		character = character:slavic_rite_cyril_1
	}
	immediate = {
		character:slavic_rite_methodius_1 = {
			save_scope_as = methodius_scope
			hidden_effect = {
				if = {
					limit = {
						has_character_modifier = pam_brothers_travel_safety_modifier
					}
					remove_character_modifier = pam_brothers_travel_safety_modifier
				}
			}
		}
		save_scope_as = moravia_liege_scope
		character:slavic_rite_cyril_1 = {
			save_scope_as = cyril_scope
		}
		title:k_papal_state.holder = {
			save_scope_as = pope_scope
		}
		scope:methodius_scope = {
			if = { # create_rite_from_type errors and does not convert if the type already exists from history
				limit = {
					NOT = { exists = rite:slavic_rite }
				}
				create_rite_from_type = { type = slavic_rite convert = yes }
			}
			set_character_rite = rite:slavic_rite
		}
		if = { # Player chose to send a gift in slavic_rite.0001
			limit = {
				has_variable = moravian_mission_gift_sent
			}
			reverse_add_opinion = {
				target = scope:pope_scope
				modifier = pious_opinion
				opinion = 25
			}
			custom_tooltip = slavic_rite_moravian_mission_gift_received_tooltip
		}
		custom_tooltip = {
			create_title_and_vassal_change = { # Create the new ecclesiastical title
				type = created
				save_scope_as = change
				add_claim_on_loss = no
			}
			capital_province = {
				save_scope_as = region_capital
			  	create_clerical_region = {
					domicile_type = ecclesiastical_domicile
					name = slavic_rite_ecclesiastical_title_name
					save_scope_as = new_title
				}
			  	scope:new_title ?= {
			  		set_coa = root.primary_title
			  		set_color_from_title = root.primary_title
			  		set_landless_title = yes
			  		set_no_automatic_claims = yes
			  		change_title_holder = {
			  			holder = scope:methodius_scope
			  			change = scope:change
			  		}
			  		resolve_title_and_vassal_change = scope:change
			  	}
			}
			text = slavic_rite_ecclesiastical_title_creation_tooltip
		}
		hidden_effect = {
			scope:methodius_scope = {
				change_government = ecclesiastical_government
			}
			if = {
				limit = {
					any_character_situation = {
						situation_type = the_christian_church
						situation_top_has_catalyst = catalyst_the_christian_church_clerical_region_created
					}
				}
				situation:the_christian_church = {
					trigger_situation_catalyst = catalyst_the_christian_church_clerical_region_created
				}
			}
		}
		
		set_global_variable = methodius_returns_from_rome_var
	}

	option = { # Convert to the rite
		name = slavic_rite.0005.a
		set_character_rite_with_conversion = rite:slavic_rite
		reverse_add_opinion = {
			target = scope:methodius_scope
			modifier = pious_opinion
			opinion = 75
		}
		add_piety = major_piety_gain

		ai_chance = {
			base = 1
		}
	}

	option = { # Don't convert, but raise a minor statue
		trigger = {
			gold >= medium_gold_value
		}
		name = slavic_rite.0005.b
		capital_province = {
			add_province_modifier = {
				modifier = pam_cyril_statue_modifier
				years = 20
			}
		}
		remove_short_term_gold = medium_gold_value
		reverse_add_opinion = {
			target = scope:methodius_scope
			modifier = pleased_opinion
			opinion = 25
		}

		ai_chance = {
			base = 1
		}
	}

	option = { # Good luck bud
		name = slavic_rite.0005.c
		add_piety = minor_piety_gain

		ai_chance = {
			base = 1
		}
	}

	after = {
		custom_tooltip = { # Start the mass conversion projects
			text = slavic_rite_mass_conversion_tooltip
			set_variable = no_great_project_cost
			every_vassal = {
				limit = {
					primary_title = { tier = tier_duchy }
				}
				capital_province = {
					plan_great_project = {
						founder = root
						great_project_type = pam_mass_rite_conversion
						save_scope_as = slavic_rite_mass_conversion_project
					}
				}
				scope:slavic_rite_mass_conversion_project ?= { # Convert the realm to the Slavic Rite regardless of the founder's own rite (TIT-74341)
					set_variable = {
						name = target_rite
						value = rite:slavic_rite
					}
				}
				random_great_project = {
					limit = {
						great_project_owner = root
						great_project_construction_phase_is = planned
						great_project_type = pam_mass_rite_conversion
					}
					random_contribution = {
						limit = {
							contribution_is_funded = no
							contribution_is_required = yes
						}
						save_scope_as = contribution_to_fund
					}
				}
				if = { # The scope is only saved when the effects actually run; guard so the tooltip/description pass doesn't error on it
					limit = {
						exists = scope:contribution_to_fund
					}
					fund_great_project_contribution = {
						contribution = scope:contribution_to_fund
						cost = no
						check_can_contribute = no
					}
				}
			}
			remove_variable = no_great_project_cost
		}
	}
}

slavic_rite.0007 = { # All counties under the new archdiocese are successfully converted to Slavic Rite, and the story ends
	type = character_event
	title = slavic_rite.0007.t
	desc = slavic_rite.0007.desc
	window = big_event_window
	theme = christian
	override_background = {
		reference = pam_simple_church_western_interior
	}
	right_portrait = {
		character = root
		camera = camera_herder_left
		animation = prayer_standing
	}
	center_portrait = {
		character = scope:archdiocese_holder
		camera = camera_event_horse_right_facing_left_close
		animation = prayer_standing_eyes
	}
	immediate = {
		if = { # Realm priest's superior is normally the archbishop
			limit = {
				exists = cp:councillor_court_chaplain.superior
			}
			cp:councillor_court_chaplain.superior = {
				save_scope_as = archdiocese_holder
			}
		}
		else_if = { # Fall back to the archdiocese holder when the realm priest has no superior (e.g. no Head of Faith yet)
			limit = {
				exists = title:c_brno.clerical_region_title.holder
			}
			title:c_brno.clerical_region_title.holder = {
				save_scope_as = archdiocese_holder
			}
		}
		else = {
			cp:councillor_court_chaplain = {
				save_scope_as = archdiocese_holder
			}
		}
		if = {
			limit = {
				NOT = { has_relation_best_friend = scope:archdiocese_holder }
			}
			set_relation_best_friend = { reason = best_friend_sworn copy_reason = friend target = scope:archdiocese_holder }
		}
		random_character_tenet = {
			knowledge_filter = unknown
			save_scope_as = discovered_tenet
		}
		add_legitimacy = major_legitimacy_gain
	}

	option = { # Clergy boost
		name = slavic_rite.0007.a
		add_piety = massive_piety_gain
		add_character_modifier = {
			modifier = pam_slavic_rite_clergy_opinion_modifier
			years = 15
		}
		add_known_tenet = scope:discovered_tenet

		stress_and_fulfillment_impact = {
			zealous = major_stress_impact_loss
			ambitious = major_stress_impact_loss
		}
		ai_chance = {
