petition_head_of_faith.0601 = {
	type = character_event
	title = petition_head_of_faith.0601.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = { exists = var:petition_head_of_faith_recipient }
				}
				desc = petition_head_of_faith.0601.desc.no_head
			}
			triggered_desc = {
				trigger = {
					var:petition_head_of_faith_recipient = { is_alive = no }
				}
				desc = petition_head_of_faith.0601.desc.head_dead
			}
			triggered_desc = {
				trigger = {
					var:petition_head_of_faith_recipient = { is_imprisoned = yes }
				}
				desc = petition_head_of_faith.0601.desc.head_imprisoned
			}
			triggered_desc = {
				trigger = {
					OR = {
						NOT = { petition_head_of_faith_has_religious_authority_trigger = yes }
						AND = {
							exists = var:petition_head_of_faith_recipient
							NOT = {
								petition_head_of_faith_religious_authority_is_trigger = { CHARACTER = var:petition_head_of_faith_recipient }
							}
						}
					}
				}
				desc = petition_head_of_faith.0601.desc.head_changed
			}
			triggered_desc = {
				trigger = {
					exists = var:petition_head_of_faith_recipient
					is_at_war_with = var:petition_head_of_faith_recipient
				}
				desc = petition_head_of_faith.0601.desc.at_war
			}
			triggered_desc = {
				trigger = {
					var:petition_head_of_faith_type = flag:repentance
					NOT = { has_trait = excommunicated }
				}
				desc = petition_head_of_faith.0601.desc.repentance_resolved
			}
			triggered_desc = {
				trigger = {
					var:petition_head_of_faith_type = flag:repentance
					has_character_modifier = recent_excommunication
				}
				desc = petition_head_of_faith.0601.desc.repentance_too_soon
			}
			triggered_desc = {
				trigger = {
					var:petition_head_of_faith_type = flag:request_excommunication
					exists = var:petition_head_of_faith_target_character
					var:petition_head_of_faith_target_character = { is_alive = no }
				}
				desc = petition_head_of_faith.0601.desc.target_dead
			}
			triggered_desc = {
				trigger = {
					var:petition_head_of_faith_type = flag:request_excommunication
					exists = var:petition_head_of_faith_target_character
					var:petition_head_of_faith_target_character = { faith != root.faith }
				}
				desc = petition_head_of_faith.0601.desc.target_wrong_faith
			}
			triggered_desc = {
				trigger = {
					var:petition_head_of_faith_type = flag:request_excommunication
					exists = var:petition_head_of_faith_target_character
					var:petition_head_of_faith_target_character = { highest_held_title_tier < tier_kingdom }
				}
				desc = petition_head_of_faith.0601.desc.target_too_low
			}
			triggered_desc = {
				trigger = {
					var:petition_head_of_faith_type = flag:request_excommunication
					exists = var:petition_head_of_faith_target_character
					var:petition_head_of_faith_target_character = { has_trait = excommunicated }
				}
				desc = petition_head_of_faith.0601.desc.target_already_excommunicated
			}
			desc = petition_head_of_faith.0601.desc.fallback
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:petition_hof_piety_refunded }
				desc = petition_head_of_faith.0601.desc.refund.piety
			}
			desc = petition_head_of_faith.0601.desc.refund.none
		}
	}
	theme = realm

	left_portrait = { character = root animation = personality_rational }
	right_portrait = {
		character = var:petition_head_of_faith_recipient
		animation = personality_honorable
		trigger = { exists = var:petition_head_of_faith_recipient }
	}

	immediate = {
		petition_head_of_faith_save_scopes_effect = yes
		# We save first because the refund effect clears petition_head_of_faith_reserved_piety_value
		if = {
			limit = { has_variable = petition_head_of_faith_reserved_piety_value }
			save_scope_value_as = {
				name = petition_hof_piety_refunded
				value = yes
			}
		}
		petition_head_of_faith_refund_reserved_costs_effect = yes
	}

	option = {
		name = petition_head_of_faith.0601.a
		current_travel_plan ?= {
			if = {
				limit = { can_cancel = yes }
				cancel_travel_plan = yes
			}
		}
	}

	after = {
		petition_head_of_faith_cleanup_effect = yes
	}
}

petition_head_of_faith.9000 = {
	hidden = yes
	immediate = {
		current_travel_plan ?= { pause_travel_plan = yes }
		if = {
			limit = { petition_head_of_faith_valid_trigger = no }
			trigger_event = petition_head_of_faith.0601
		}
		else_if = {
			limit = {
				var:petition_head_of_faith_recipient = {
					has_royal_court = yes
				}
			}
			trigger_event = petition_head_of_faith.3000
		}
		else = {
			trigger_event = petition_head_of_faith.1000
		}
	}
}

scripted_trigger petition_head_of_faith_arrival_recipient_sinful_trigger = {
	var:petition_head_of_faith_recipient = {
		num_sinful_traits >= 1
	}
}

scripted_trigger petition_head_of_faith_arrival_recipient_poor_health_trigger = {
	var:petition_head_of_faith_recipient = {
		OR = {
			has_trait = infirm
			health <= poor_health
			has_any_debilitating_illness = yes
		}
	}
}

scripted_trigger petition_head_of_faith_arrival_recipient_reputable_trigger = {
	var:petition_head_of_faith_recipient = {
		OR = {
			num_virtuous_traits > num_sinful_traits
			num_sinful_traits = 0
		}
	}
}

scripted_trigger petition_head_of_faith_arrival_recipient_virtuous_trigger = {
	var:petition_head_of_faith_recipient = {
		num_virtuous_traits >= 1
	}
}

# Arrival (no royal court)
petition_head_of_faith.1000 = {
	type = character_event
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { var:petition_head_of_faith_type = flag:repentance }
				desc = petition_head_of_faith.1000.t.repentance
			}
			triggered_desc = {
				trigger = { var:petition_head_of_faith_type = flag:request_excommunication }
				desc = petition_head_of_faith.1000.t.excommunication
			}
		}
	}
	desc = {
		desc = petition_head_of_faith.1000.desc.intro
		triggered_desc = {
			trigger = {
				petition_head_of_faith_arrival_recipient_sinful_trigger = yes
				exists = scope:petition_hof_gravest_sin
			}
			desc = petition_head_of_faith.1000.desc.recipient_sinful
		}
		triggered_desc = {
			trigger = {
				petition_head_of_faith_arrival_recipient_virtuous_trigger = yes
				exists = scope:petition_hof_gravest_virtue
			}
			desc = petition_head_of_faith.1000.desc.recipient_virtuous
		}
		triggered_desc = {
			trigger = {
				petition_head_of_faith_arrival_recipient_poor_health_trigger = yes
				petition_head_of_faith_arrival_recipient_reputable_trigger = yes
			}
			desc = petition_head_of_faith.1000.desc.health_noble
		}
		triggered_desc = {
			trigger = {
				petition_head_of_faith_arrival_recipient_poor_health_trigger = yes
				NOT = { petition_head_of_faith_arrival_recipient_reputable_trigger = yes }
			}
			desc = petition_head_of_faith.1000.desc.health_excess
		}
		desc = petition_head_of_faith.1000.desc.address
		first_valid = {
			triggered_desc = {
				trigger = { var:petition_head_of_faith_type = flag:repentance }
				desc = petition_head_of_faith.1000.desc.repentance
			}
			triggered_desc = {
				trigger = { var:petition_head_of_faith_type = flag:request_excommunication }
				desc = petition_head_of_faith.1000.desc.excommunication
			}
		}
		# Disposition indicators in arrival desc (favorable/hostile/skeptical/neutral).
		first_valid = {
			triggered_desc = {
				trigger = { petition_head_of_faith_disposition_favorable_trigger = yes }
				desc = petition_head_of_faith.1000.desc.disposition_favorable
			}
			triggered_desc = {
				trigger = { petition_head_of_faith_disposition_neutral_trigger = yes }
				desc = petition_head_of_faith.1000.desc.disposition_neutral
			}
			triggered_desc = {
				trigger = { petition_head_of_faith_disposition_skeptical_trigger = yes }
				desc = petition_head_of_faith.1000.desc.disposition_skeptical
			}
			triggered_desc = {
				trigger = { petition_head_of_faith_disposition_hostile_trigger = yes }
				desc = petition_head_of_faith.1000.desc.disposition_hostile
			}
		}
		desc = petition_head_of_faith.1000.desc.conclusion
	}
	theme = realm
	left_portrait = { character = root animation = throne_room_bow_1 }
	right_portrait = { character = var:petition_head_of_faith_recipient animation = holding_hu }
	override_background = { reference = throne_room_scope }

	immediate = {
		remove_character_flag = petition_head_of_faith_in_courtroom
		petition_head_of_faith_save_scopes_effect = yes
		if = {
			limit = { petition_head_of_faith_arrival_recipient_sinful_trigger = yes }
			scope:recipient = {
				random_character_trait = {
					limit = {
						scope:recipient.rite = {
							trait_is_sin_rite = prev
						}
					}
					save_scope_as = petition_hof_gravest_sin
				}
			}
		}
		if = {
			limit = { petition_head_of_faith_arrival_recipient_virtuous_trigger = yes }
			scope:recipient = {
				random_character_trait = {
					limit = {
						scope:recipient.rite = {
							trait_is_virtue_rite = prev
						}
					}
					save_scope_as = petition_hof_gravest_virtue
				}
			}
		}
		var:petition_head_of_faith_recipient = {
			save_scope_as = background_throne_room_scope
		}
	}

	# All petition types: route to argument duel (AI) or human Head of Faith decision event.
	option = {
		name = petition_head_of_faith.1000.a.argue
		custom_tooltip = petition_head_of_faith.1000.a.argue_tt
		if = {
			limit = { var:petition_head_of_faith_recipient = { is_ai = yes } }
			# Route to type-specific argument event
			if = {
				limit = { var:petition_head_of_faith_type = flag:request_excommunication }
				trigger_event = petition_head_of_faith.2000
			}
			else_if = {
				limit = { var:petition_head_of_faith_type = flag:repentance }
				trigger_event = petition_head_of_faith.5000
			}
		}
		else = {
			var:petition_head_of_faith_recipient = {
				set_variable = {
					name = petition_head_of_faith_petitioner
					value = root
				}
				if = {
					limit = { root.var:petition_head_of_faith_type = flag:repentance }
					trigger_event = petition_head_of_faith.1310
				}
				else_if = {
					limit = { root.var:petition_head_of_faith_type = flag:request_excommunication }
					trigger_event = petition_head_of_faith.1320
				}
			}
		}
	}
}
