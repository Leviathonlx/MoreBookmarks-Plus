namespace = natural_primitivism

natural_primitivism.0001 = {
	type = character_event
	title = natural_primitivism.0001.t
	desc = {
		# The decree names its audience: the council when the featured character sits on it, otherwise the whole court
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:naturism_councillor
					scope:naturism_councillor = { is_councillor_of = root }
				}
				desc = natural_primitivism.0001.desc.decree.council
			}
			desc = natural_primitivism.0001.desc.decree.court
		}
		# The reaction: a supporter, an ashamed courtier, or nobody in particular
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:naturism_councillor
					scope:naturism_councillor = { natural_primitivism_supports_naturism_trigger = yes }
				}
				desc = natural_primitivism.0001.desc.reaction.supportive
			}
			triggered_desc = {
				trigger = { exists = scope:naturism_councillor }
				desc = natural_primitivism.0001.desc.reaction.ashamed
			}
			desc = natural_primitivism.0001.desc.reaction.dissent
		}
	}
	theme = faith
	left_portrait = {
		character = root
		animation = personality_zealous
	}
	right_portrait = {
		trigger = { exists = scope:naturism_councillor }
		character = scope:naturism_councillor
		triggered_animation = {
			trigger = { natural_primitivism_supports_naturism_trigger = yes }
			animation = ecstasy
		}
		animation = shame
	}

	immediate = {
		if = {
			limit = {
				any_councillor = {
					is_available_adult = yes
					natural_primitivism_supports_naturism_trigger = yes
				}
			}
			random_courtier = {
				limit = {
					is_available_adult = yes
					is_councillor_of = root
					natural_primitivism_supports_naturism_trigger = yes
				}
				save_scope_as = naturism_councillor
			}
		}
		else_if = {
			limit = {
				any_courtier = {
					is_available_adult = yes
					is_councillor_of = root
				}
			}
			random_courtier = {
				limit = {
					is_available_adult = yes
					is_councillor_of = root
				}
				save_scope_as = naturism_councillor
			}
		}
		else_if = {
			limit = {
				any_courtier = {
					is_available_adult = yes
					natural_primitivism_supports_naturism_trigger = yes
				}
			}
			random_courtier = {
				limit = {
					is_available_adult = yes
					natural_primitivism_supports_naturism_trigger = yes
				}
				save_scope_as = naturism_councillor
			}
		}
		else_if = {
			limit = {
				any_courtier = { is_available_adult = yes }
			}
			random_courtier = {
				limit = { is_available_adult = yes }
				save_scope_as = naturism_councillor
			}
		}
	}

	option = {
		name = natural_primitivism.0001.a
		flavor = natural_primitivism.0001.a.flavor
		natural_primitivism_promote_court_naturism_effect = yes
		ai_chance = {
			base = 100
		}
	}
}

natural_primitivism.0003 = {
	type = character_event
	title = natural_primitivism.0003.t
	desc = natural_primitivism.0003.desc
	theme = faith
	left_portrait = {
		character = root
		animation = worry
	}
	right_portrait = {
		trigger = { exists = scope:naturism_liege }
		character = scope:naturism_liege
		animation = personality_zealous
	}

	trigger = {
		is_available_adult = yes
		is_ai = no
		exists = scope:naturism_liege
		scope:naturism_liege = { has_character_flag = natural_primitivism_promoted_naturism_active }
		NOT = { has_character_modifier = natural_primitivism_court_naturism_modifier }
		NOT = { has_character_flag = natural_primitivism_rejected_court_naturism }
	}

	option = {
		name = natural_primitivism.0003.a
		flavor = natural_primitivism.0003.a.flavor
		natural_primitivism_apply_court_naturism_to_character_effect = yes
		ai_chance = {
			base = 100
		}
	}
	option = {
		name = natural_primitivism.0003.b
		flavor = natural_primitivism.0003.b.flavor
		add_stress = medium_stress_gain
		add_character_flag = natural_primitivism_rejected_court_naturism
		hidden_effect = {
			scope:naturism_liege = {
				add_opinion = {
					target = root
					modifier = natural_primitivism_rejected_court_naturism_ruler_opinion
				}
				every_courtier = {
					limit = {
						is_available_adult = yes
						NOT = { this = root }
						OR = {
							has_character_modifier = natural_primitivism_court_naturism_modifier
							natural_primitivism_supports_naturism_trigger = yes
						}
					}
					add_opinion = {
						target = root
						modifier = natural_primitivism_rejected_court_naturism_court_opinion
					}
				}
			}
		}
		custom_tooltip = natural_primitivism.0003.b.tt
		ai_chance = {
			base = 0
		}
	}
}
