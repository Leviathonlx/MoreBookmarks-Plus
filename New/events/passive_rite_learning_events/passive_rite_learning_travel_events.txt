namespace = rite_travel_events

### Fascinating Person
rite_travel_events.1000 = {
	type = character_event
	title = rite_travel_events.1000.t
	desc = rite_travel_events.1000.desc
	theme = travel
	override_background = ep3_campfire
	override_effect_2d = { reference = smoke }

	left_portrait = {
		character = root
		animation = interested
		camera = camera_event_center_pointing_right
	}
	right_portrait = {
		character = scope:passive_learn_source
		animation = debating
		camera = camera_event_right_pointing_right_scheme
	}

	cooldown = { years = 5 }

	trigger = {
		is_available_travelling_adult = yes
		is_location_valid_for_travel_event_on_land = yes
		any_pool_character = {
			province = root.location
			is_available_healthy_ai_adult = yes
			faith != root.faith
			is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
		}
	}

	immediate = {
		random_pool_character = {
			province = root.location
			limit = {
				is_available_healthy_ai_adult = yes
				faith != root.faith
				is_valid_passive_learning_source_for_learner_trigger = { LEARNER = root }
			}
			save_scope_as = passive_learn_source
		}
		scope:passive_learn_source.rite = { save_scope_as = passive_learn_rite }
	}

	# Study them
	option = {
		name = rite_travel_events.1000.a
		add_known_rite = scope:passive_learn_rite
		worldly_type_option_effect = yes
		send_interface_message = {
			type = msg_passive_learned_rite
			desc = msg_passive_learned_rite_abroad_desc
			left_icon = scope:passive_learn_source
			right_icon = scope:passive_learn_rite
		}
		stress_and_fulfillment_impact = {
			base = minor_stress_impact_loss
			humble = minor_stress_impact_loss
		}
	}

	# Invite them into your entourage
	option = {
		name = rite_travel_events.1000.b
		add_known_rite = scope:passive_learn_rite
		worldly_type_option_effect = yes
		send_interface_message = {
			type = msg_passive_learned_rite
			desc = msg_passive_learned_rite_abroad_desc
			left_icon = scope:passive_learn_source
			right_icon = scope:passive_learn_rite
		}
		scope:passive_learn_source = {
			add_to_court_and_entourage_effect = yes
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 50
			}
		}
		stress_and_fulfillment_impact = {
			zealous = minor_stress_impact_gain
			paranoid = minor_stress_impact_gain
		}
	}

	# Decline
	option = {
		name = rite_travel_events.1000.c
		add_piety = minor_piety_gain
		pious_type_option_effect = yes
		stress_and_fulfillment_impact = {
			humble = minor_stress_impact_gain
		}
	}
}

### Revelation
rite_travel_events.2000 = {
	type = character_event
	title = rite_travel_events.2000.t
	desc = rite_travel_events.2000.desc
	theme = travel

	left_portrait = {
		character = root
		animation = divine_ecstasy
	}
	right_portrait = {
		character = scope:passive_learn_source
		animation = preaching
		camera = camera_event_very_right_fov35
	}

	cooldown = { years = 100 }

	trigger = {
		is_available_travelling_adult = yes
		is_location_valid_for_travel_event_on_land = yes
		any_pool_character = {
			province = root.location
			faith != root.faith
			is_available_healthy_ai_adult = yes
			save_temporary_scope_as = revelation_source
			root = {
				any_character_tenet = {
					knowledge_filter = unknown
					passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:revelation_source.rite }
					passive_rite_learning_tenet_resonates_with_player_trigger = { CHARACTER = root }
					count >= 1
				}
			}
		}
	}

	immediate = {
		location = {
			save_scope_as = location
		}
		random_pool_character = {
			province = root.location
			limit = {
				faith != root.faith
				is_available_adult = yes
				save_temporary_scope_as = revelation_source
				root = {
					any_character_tenet = {
						knowledge_filter = unknown
						passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:revelation_source.rite }
						passive_rite_learning_tenet_resonates_with_player_trigger = { CHARACTER = root }
						count >= 1
					}
				}
			}
			save_scope_as = passive_learn_source
		}
		random_character_tenet = {
			knowledge_filter = unknown
			limit = {
				passive_rite_learning_candidate_tenet_is_core_trigger = { RITE = scope:passive_learn_source.rite }
				passive_rite_learning_tenet_resonates_with_player_trigger = { CHARACTER = root }
			}
			save_scope_as = passive_learn_tenet
		}
	}

	option = {
		name = rite_travel_events.2000.a
		add_known_tenet = scope:passive_learn_tenet
		force_set_personal_tenet = scope:passive_learn_tenet
		change_spiritual_fulfillment = {
			value = 50
		}
	}

	option = {
		name = rite_travel_events.2000.b
		add_known_tenet = scope:passive_learn_tenet
		add_piety = minor_piety_gain
	}
}
