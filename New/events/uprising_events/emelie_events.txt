# Random starter event for the Blot-Sven storyline. You get offered the throne of Sweden/your Christian liege if you perform the Asatru blot.
blotsven.0001 = {
	type = character_event
	title = blotsven.0001.t
	desc = blotsven.0001.desc
	theme = faith
	override_background = { reference = fp1_tribal_temple }
	left_portrait = {
		character = root
		animation = interested
		camera = camera_event_center_background_crowd_left
	}
	right_portrait = {
		character = cp:councillor_court_chaplain
		animation = holding_staff
		camera = camera_event_right_away
	}

	trigger = {
		current_date >= 1066.9.15
		current_date <= 1080.1.1
		exists = title:k_sweden.holder
		# Cannot get this again if you already have it but haven't acted on it
		NOT = { has_character_flag = blotsven }
		# DLC check so you can actually do blot activity.
		has_fp1_dlc_trigger = yes
		# Standard checks.
		is_available_even_at_war_adult = yes
		is_landed = yes
		highest_held_title_tier >= tier_county
		# Should be North Germanic, or the flavour won't make sense.
		culture = { has_cultural_pillar = heritage_north_germanic }
		# Must be able to do a blot
		NOT = { is_activity_type_on_cooldown = activity_gruesome_festival }
		faith = {
			has_doctrine_parameter = gruesome_festivals_active
		}
		character:eskil = {
			is_alive = yes
		}
		this != top_liege
		top_liege = {
			has_title = title:k_sweden
			faith = { religion_tag = christianity_religion }
		}
	}

	immediate = {
		cp:councillor_court_chaplain = { save_scope_as = chaplain }
	}

	option = { # Start planning blot
		name = blotsven.0001.a
		add_character_flag = {
			flag = blotsven
			# When you plan a blot in the future, it will check for this flag, and launch the events if you have it
		}
		custom_tooltip = {
			text = blotsven.0001.a.tt
		}
		# AI should not engage with this event
		ai_chance = {
			base = 0
		}
	}
	option = { # Do nothing
		name = blotsven.0001.b
		custom_tooltip = {
			text = blotsven.0001.b.tt
		}
		# Nothing happens
		# AI should not engage with this event
		ai_chance = {
			base = 100
		}
	}
}

# Eskil shows up
blotsven.0002 = {
	type = activity_event
	title = blotsven.0002.t
	desc = blotsven.0002.desc
	theme = faith
	override_background = { reference = activity_witch_ritual }
	override_effect_2d = { reference = rain }
	left_portrait = {
		character = root
		animation = aggressive_axe
		camera = camera_event_center_mostly_away_right
	}
	right_portrait = {
		character = character:eskil
		animation = preaching_no_book
		camera = camera_event_center_background_crowd_left
		outfit_tags = { eskil_outfit }
	}

	trigger = {
		character:eskil = {
			is_alive = yes
		}
	}

	immediate = {
		character:eskil = { save_scope_as = eskil }
		remove_character_flag = blotsven
	}

	option = { # Throw rock
		name = blotsven.0002.a
		flavor = blotsven.0002.a.tt
		duel = {
			skill = prowess
			target = scope:eskil
			# Success
			50 = {
				compare_modifier = {
					value = prowess
					multiplier = 2
					min = -49
				}
				scope:activity = {
					add_activity_log_entry = {
						key = gruesome_activity_eskil_slain
						character = scope:eskil
						show_in_conclusion = yes
					}
				}
				scope:eskil = {
					death = {
						death_reason = death_execution
						killer = root
					}
					add_trait = saint
				}
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = blotsven.0002.toast.k.tt
					left_icon = root
					right_icon = scope:eskil
				}
				add_pressed_claim = title:k_sweden
				add_piety = minor_piety_gain
				stress_and_fulfillment_impact = {
					zealous = minor_stress_impact_loss
				}
				give_nickname = nick_blotsven
			}
			# Failure
			50 = {
				compare_modifier = {
					value = prowess
					multiplier = -2
					min = -49
				}
			scope:eskil = {
				set_relation_potential_rival = root
				add_opinion = {
					target = root
					modifier = murder_personal_grudge_opinion
					opinion = -50
				}
			}
			trigger_event = blotsven.0003
			}
		}
	}
	option = { # Throw axe
		name = blotsven.0002.b
		flavor = blotsven.0002.b.tt
		duel = {
			skill = prowess
			target = scope:eskil
			50 = {
				compare_modifier = {
					value = prowess
					multiplier = 3.5
					min = -49
				}
				add_stress = minor_stress_loss
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = blotsven.0002.toast.m.tt
					left_icon = root
					right_icon = scope:eskil
				}
				trigger_event = blotsven.0003
				scope:eskil = {
					add_trait = maimed
					add_opinion = {
						target = root
						modifier = maimed_me
						opinion = -50
					}
					set_relation_potential_rival = root
				}
			}
			50 = {
				compare_modifier = {
					value = prowess
					multiplier = -3.5
					min = -49
				}
			scope:eskil = {
				set_relation_potential_rival = root
				add_opinion = {
					target = root
					modifier = murder_personal_grudge_opinion
					opinion = -50
				}
			}
			trigger_event = blotsven.0003
			}
		}
	}
	option = { # Stop the blot and convert
		name = blotsven.0002.c
		flavor = blotsven.0002.c.tt
		set_character_faith = scope:eskil.faith
		add_character_flag = { flag = changed_religion_blot_flag }
		scope:eskil = {
			add_opinion = {
				target = root
				modifier = friendliness_opinion
				opinion = 30
			}
		}
		every_vassal = { # Asatru vassals will dislike this
			limit = {
				faith = faith:norse_pagan
			}
			add_opinion = {
				target = prev
				modifier = minority_disapproves_of_conversion_opinion
				opinion = -30
			}
		}
	}
}

# Eskil survived and is pissed
blotsven.0003 = {
	type = activity_event
	title = blotsven.0003.t
	desc = blotsven.0003.desc
	theme = faith
	override_background = { reference = ep2_forest_managed }
	override_effect_2d = { reference = rain }
	left_portrait = {
		character = root
		animation = menacing
		camera = camera_event_horse_left
	}
	right_portrait = {
		character = character:eskil
		animation = aggressive_unarmed
		camera = camera_event_right_away
		outfit_tags = { eskil_outfit }
	}
	option = { # Kill him again
		name = blotsven.0003.a
		flavor = blotsven.0003.a.tt
		scope:activity = {
			add_activity_log_entry = {
				key = gruesome_activity_eskil_slain
				character = scope:eskil
				show_in_conclusion = yes
			}
		}
		scope:eskil = {
			set_relation_rival = root
			death = {
				death_reason = death_execution
				killer = root
			}
			add_trait = saint
		}
		add_pressed_claim = title:k_sweden
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = blotsven.0003.toast.tt
			tooltip = blotsven.0003.toast.desc.tt
			left_icon = root
			right_icon = scope:eskil
		}
		add_stress = minor_stress_loss
	}
	option = { # Listen to him and convert
		name = blotsven.0003.b
		flavor = blotsven.0003.b.tt
		set_character_faith = scope:eskil.faith
		add_character_flag = { flag = changed_religion_blot_flag }
		every_vassal = { # Asatru vassals will dislike this
			limit = {
				faith = faith:norse_pagan
			}
			add_opinion = {
				target = prev
				modifier = unwanted_religious_conversion_opinion
				opinion = -30
			}
		}
	}
}
