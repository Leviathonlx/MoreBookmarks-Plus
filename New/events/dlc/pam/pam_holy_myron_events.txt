# Spent vessel: we destroy after the spender (trigger_event defers), and nothing may follow the destroy
pam_holy_myron.0002 = {
	hidden = yes

	trigger = {
		exists = scope:pam_holy_myron_spent_vessel
	}

	immediate = {
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = pam_holy_myron_vessel_spent_toast
			left_icon = root
			custom_tooltip = pam_holy_myron_vessel_spent_tt
			show_as_tooltip = {
				destroy_artifact = scope:pam_holy_myron_spent_vessel
			}
		}
		destroy_artifact = scope:pam_holy_myron_spent_vessel
	}
}

# We count the journey's legs instead of checking locations, so the route cannot be looped
pam_holy_myron.0010 = {
	hidden = yes

	trigger = {
		has_variable = pam_holy_myron_journey_tier
	}

	immediate = {
		save_scope_as = myron_chaplain
		location = { save_scope_as = myron_waypoint }
		change_variable = {
			name = pam_holy_myron_legs
			add = 1
		}
		if = {
			limit = {
				var:pam_holy_myron_legs < var:pam_holy_myron_expected_legs
			}
			random_list = {
				70 = { # Steady progress
					var:pam_holy_myron_liege ?= {
						trigger_event = {
							id = pam_holy_myron.0011
						}
					}
				}
				15 = { # Bandits demand a toll - a fighting cleric is a poor mark, and once per journey
					trigger = {
						NOT = { has_variable = pam_holy_myron_ransom_fired }
					}
					modifier = {
						factor = 0.5
						prowess >= decent_skill_rating
					}
					modifier = {
						factor = 0.5
						prowess >= high_skill_rating
					}
					set_variable = {
						name = pam_holy_myron_ransom_fired
						years = 30
					}
					var:pam_holy_myron_liege ?= {
						trigger_event = {
							id = pam_holy_myron.0012
						}
					}
				}
				15 = { # A wasting fever - the learned know which wells to drink from, and once per journey
					trigger = {
						can_contract_disease_trigger = { DISEASE = ill }
						NOT = { has_variable = pam_holy_myron_fever_fired }
					}
					modifier = {
						factor = 0.5
						learning >= decent_skill_rating
					}
					modifier = {
						factor = 0.5
						learning >= high_skill_rating
					}
					set_variable = {
						name = pam_holy_myron_fever_fired
						years = 30
					}
					contract_disease_effect = {
						DISEASE = ill
						TREATMENT_EVENT = no
					}
					var:pam_holy_myron_liege ?= {
						trigger_event = {
							id = pam_holy_myron.0013
						}
					}
				}
			}
		}
		else = {
			# No living payer and no liege: cleanup, the chaplain does not consecrate
			if = {
				limit = {
					exists = var:pam_holy_myron_liege
					var:pam_holy_myron_liege = { is_alive = yes }
				}
				var:pam_holy_myron_liege = {
					trigger_event = {
						id = pam_holy_myron.0100
						days = 2
					}
				}
			}
			else_if = {
				limit = { exists = liege }
				liege = {
					trigger_event = {
						id = pam_holy_myron.0100
						days = 2
					}
				}
			}
			else = {
				pam_holy_myron_journey_cleanup_effect = yes
			}
		}
	}
}

# The chaplain returns; the new Myron is consecrated and the ampulla created
pam_holy_myron.0100 = {
	type = character_event
	title = pam_holy_myron.0100.t
	desc = {
		desc = pam_holy_myron.0100.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:myron_chaplain = { has_variable = pam_holy_myron_robbed }
				}
				desc = pam_holy_myron.0100.desc_robbed
			}
			desc = pam_holy_myron.0100.desc_full
		}
	}
	theme = christian
	override_background = {
		reference = temple
	}

	left_portrait = {
		character = root
		triggered_animation = {
			trigger = {
				OR = {
					has_trait = zealous
					ai_zeal >= low_positive_ai_value
				}
			}
			animation = personality_zealous
		}
		animation = happiness
		camera = camera_event_very_left
	}
	right_portrait = {
		character = scope:myron_chaplain
		animation = situla
	}
	artifact = {
		target = scope:newly_created_artifact
		position = lower_center_portrait
	}

	trigger = {
		scope:myron_chaplain ?= { is_alive = yes }
	}

	immediate = {
		if = {
			limit = {
				scope:myron_chaplain = { var:pam_holy_myron_journey_tier = 1 }
			}
			save_scope_value_as = {
				name = myron_rarity
				value = flag:common
			}
		}
		else_if = {
			limit = {
				scope:myron_chaplain = { var:pam_holy_myron_journey_tier = 2 }
			}
			if = {
				limit = {
					scope:myron_chaplain = { has_variable = pam_holy_myron_robbed }
				}
				save_scope_value_as = {
					name = myron_rarity
					value = flag:common
				}
			}
			else = {
				save_scope_value_as = {
					name = myron_rarity
					value = flag:masterwork
				}
			}
		}
		else_if = {
			limit = {
				scope:myron_chaplain = { var:pam_holy_myron_journey_tier = 3 }
			}
			if = {
				limit = {
					scope:myron_chaplain = { has_variable = pam_holy_myron_robbed }
				}
				save_scope_value_as = {
					name = myron_rarity
					value = flag:masterwork
				}
			}
			else = {
				save_scope_value_as = {
					name = myron_rarity
					value = flag:famed
				}
				scope:myron_chaplain = {
					random = {
						chance = pam_holy_myron_illustrious_chance_value
						root = {
							save_scope_value_as = {
								name = myron_rarity
								value = flag:illustrious
							}
						}
					}
				}
			}
		}
		else = {
			# A missing or expired journey tier should fail to the humblest batch, not the finest.
			save_scope_value_as = {
				name = myron_rarity
				value = flag:common
			}
		}
		# We create it on the chaplain so the tooltip stays safe
		hidden_effect_new_object = {
			if = {
				limit = { scope:myron_rarity = flag:common }
				pam_create_holy_myron_artifact_effect = {
					OWNER = scope:myron_chaplain
					CREATOR = scope:myron_chaplain
					RARITY = common
				}
			}
			else_if = {
				limit = { scope:myron_rarity = flag:masterwork }
				pam_create_holy_myron_artifact_effect = {
					OWNER = scope:myron_chaplain
					CREATOR = scope:myron_chaplain
					RARITY = masterwork
				}
			}
			else_if = {
				limit = { scope:myron_rarity = flag:famed }
				pam_create_holy_myron_artifact_effect = {
					OWNER = scope:myron_chaplain
					CREATOR = scope:myron_chaplain
					RARITY = famed
				}
			}
			else_if = {
				limit = { scope:myron_rarity = flag:illustrious }
				pam_create_holy_myron_artifact_effect = {
					OWNER = scope:myron_chaplain
					CREATOR = scope:myron_chaplain
					RARITY = illustrious
				}
			}
		}
	}

	option = { # The chain of blessing remains unbroken
		name = pam_holy_myron.0100.a
		scope:newly_created_artifact = {
			set_owner = { target = root }
		}
		scope:myron_chaplain = {
			add_piety = medium_piety_gain
		}
		stress_and_fulfillment_impact = {
			zealous = medium_stress_impact_loss
			cynical = miniscule_stress_impact_gain
		}
		ai_chance = {
			base = 100
		}
	}

	after = {
		scope:myron_chaplain = {
			pam_holy_myron_journey_cleanup_effect = yes
		}
	}
}

# Funeral wake: the host may spend a charge of the Holy Myron on the deceased
pam_holy_myron.0200 = {
	type = activity_event
	title = pam_holy_myron.0200.t
	desc = pam_holy_myron.0200.desc
	theme = funeral_activity

	left_portrait = {
		character = root
		animation = situla
		camera = camera_event_very_left_activity
	}
	right_portrait = {
		character = scope:deceased
		animation = dead
	}

	cooldown = { years = 5 }

	trigger = {
		has_pam_dlc_trigger = yes
		involved_activity ?= {
			has_activity_type = activity_funeral
			activity_host = root
		}
		is_christian_trigger = yes
		pam_has_usable_holy_myron_trigger = yes
		scope:deceased ?= {
			faith.religion = root.faith.religion
			NOT = { has_dead_character_variable = pam_holy_myron_anointed }
		}
	}

	immediate = {
		pam_save_holy_myron_artifact_effect = yes
	}

	option = { # Anoint the deceased
		name = pam_holy_myron.0200.a
		trigger = {
			pam_has_usable_holy_myron_trigger = yes
		}
		pam_holy_myron_spend_warning_effect = yes
		hidden_effect = {
			pam_holy_myron_consume_charge_effect = {
				ARTIFACT = scope:pam_holy_myron_artifact
			}
			scope:deceased = {
				set_dead_character_variable = {
					name = pam_holy_myron_anointed
					value = yes
					years = 100
				}
			}
		}
		add_piety = major_piety_gain
		add_legitimacy_effect = { LEGITIMACY = medium_legitimacy_gain }
		change_spiritual_fulfillment = major_spiritual_fulfillment_value
		custom_tooltip = pam_holy_myron.0200.a.saint_tt
		stress_and_fulfillment_impact = {
			base = minor_stress_impact_loss
			zealous = minor_stress_impact_loss
		}
		ai_chance = {
			base = 25
			modifier = {
				add = 100
				scope:deceased ?= {
					OR = {
						is_close_family_of = root
						AND = {
							exists = root.dynasty
							dynasty ?= root.dynasty
						}
					}
				}
			}
			modifier = {
				factor = 2
				has_trait = zealous
			}
		}
	}

	option = { # The chrism is too precious to pour on the dead
		name = pam_holy_myron.0200.b
		custom_tooltip = pam_holy_myron.0200.b.tt
		add_prestige = miniscule_prestige_gain
		stress_and_fulfillment_impact = {
			greedy = miniscule_stress_impact_loss
		}
		ai_chance = {
			base = 100
		}
	}
}

# Chrismation of the newborn: a devotee with an ampulla may spend a charge for a blessing
# The child birth on_actions fire this, and root is the parent
pam_holy_myron.0300 = {
	type = character_event
	title = pam_holy_myron.0300.t
	desc = {
		desc = pam_holy_myron.0300.desc
		triggered_desc = {
			trigger = { exists = scope:pam_myron_twin }
			desc = pam_holy_myron.0300.desc_twin
		}
	}
	theme = christian
	override_background = {
		reference = temple
	}

	left_portrait = {
		character = root
		animation = situla
		camera = camera_event_left_to_the_left
	}
	right_portrait = {
		character = scope:myron_officiant
		animation = newborn
	}
	lower_center_portrait = {
		character = scope:child
	}
	lower_right_portrait = {
		character = scope:pam_myron_twin
	}
	artifact = {
		target = scope:pam_holy_myron_artifact
		position = lower_left_portrait
	}

	trigger = {
		scope:child ?= { is_alive = yes }
		exists = cp:councillor_court_chaplain
	}

	immediate = {
		cp:councillor_court_chaplain = {
			save_scope_as = myron_officiant
		}
		pam_save_holy_myron_artifact_effect = yes
	}

	option = { # Nothing less than the Holy Myron itself
		name = pam_holy_myron.0300.b
		trigger = {
			pam_has_usable_holy_myron_trigger = yes
			scope:child = {
				NOT = { has_character_modifier = pam_myron_chrismated_child_modifier }
			}
		}
		pam_holy_myron_spend_warning_effect = yes
		hidden_effect = {
			pam_holy_myron_consume_charge_effect = {
				ARTIFACT = scope:pam_holy_myron_artifact
			}
		}
		add_piety = minor_piety_gain
		change_spiritual_fulfillment = medium_spiritual_fulfillment_value
		scope:child = {
			add_character_modifier = pam_myron_chrismated_child_modifier
		}
		scope:pam_myron_twin ?= {
			if = {
				limit = {
					is_alive = yes
					NOT = { has_character_modifier = pam_myron_chrismated_child_modifier }
				}
				add_character_modifier = pam_myron_chrismated_child_modifier
			}
		}
		stress_and_fulfillment_impact = {
			zealous = minor_stress_impact_loss
		}
		ai_chance = {
			base = 50
			modifier = {
				add = 100
				scope:child = { is_primary_heir_of = root }
			}
		}
	}

	option = { # The blessed oil of the church suffices
		name = pam_holy_myron.0300.a
		scope:child = {
			if = {
				limit = {
					NOR = {
						has_character_modifier = pam_chrismated_child_modifier
						has_character_modifier = pam_myron_chrismated_child_modifier
					}
				}
				add_character_modifier = {
					modifier = pam_chrismated_child_modifier
					years = 16
				}
			}
		}
		scope:pam_myron_twin ?= {
			if = {
				limit = {
					is_alive = yes
					NOR = {
						has_character_modifier = pam_chrismated_child_modifier
						has_character_modifier = pam_myron_chrismated_child_modifier
					}
				}
				add_character_modifier = {
					modifier = pam_chrismated_child_modifier
					years = 16
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}
}
