scripted_trigger pam_the_christian_church_0001_valid_ai_situation_char_trigger = {
	# Basic checks.
	is_ai = yes
}

pam_the_christian_church.0001 = {
	hidden = yes
	scope = situation

	# If the situation ends, break the loop.
	trigger = { exists = situation:the_christian_church }

	immediate = {
		# Cue this event up again for the next time around.
		trigger_event = {
			id = pam_the_christian_church.0001
			years = { 3 5 }
		}
	}
}

scripted_effect save_christian_church_scopes_effect = {
	character:145292 ?= { save_scope_as = st_peter } #St. Peter
	character:saint_0002 ?= { save_scope_as = st_paul } #St. Paul
	if = {
		limit = {
			situation:the_christian_church.var:tcc_main_power_faith ?= faith:christian_faith
		}
		faith:christian_faith = { save_scope_as = christianity }
	}
	else = {
		situation:the_christian_church.var:tcc_main_power_faith ?= { save_scope_as = christianity }
	}
}
 
pam_the_christian_church.0051 = { # Fragile Unity
	type = character_event
	window = fullscreen_event
	queue_icon = "gfx/interface/icons/situation_types/christian_church.dds"
	queue_name = the_christian_church
	override_sound = { reference = "event:/DLC/CE3/SFX/Stingers/ce3_mx_sting_ChristianSituation_Instability" }
	title = pam_the_christian_church.0051.t
	desc = {
		#body text
		first_valid = {	
			triggered_desc = {
				trigger = {
					top_participant_group:the_christian_church = {
						OR = {
							participant_group_type = christian_clerical_heretic_rulers
							participant_group_type = christian_regular_heretic_rulers
						}
					}
				}
				desc = pam_the_christian_church.0051.desc.heretic_rulers 
			}
			desc = pam_the_christian_church.0051.desc.intro
		}
		#End text 
		first_valid = {	
			triggered_desc = {
				trigger = {
					top_participant_group:the_christian_church = {
						OR = {
							participant_group_type = christian_clerical_main_power_rulers
							participant_group_type = christian_regular_main_power_rulers
						}
					}
				}
				desc = pam_the_christian_church.0051.desc.western 
			}
			triggered_desc = {
				trigger = {
					top_participant_group:the_christian_church = {
						OR = {
							participant_group_type = christian_clerical_secondary_powers_rulers
							participant_group_type = christian_regular_secondary_powers_rulers
						}
					}
				}
				desc = pam_the_christian_church.0051.desc.eastern 
			}
		}
		# Appended only when Fragile Unity follows a Temporal (crown) ruling at the Investiture Council.
		triggered_desc = {
			trigger = { situation:the_christian_church ?= { var:pam_investiture_last_outcome ?= flag:temporal } }
			desc = pam_the_christian_church.0051.desc.investiture_temporal
		}
	}
	theme = christian
	cooldown = { days = 1 }
	trigger = { # we need to prevent this from firing for all players when a new player joins in MP
		is_ai = no
		is_player_tutorial_character = no
 	}
 	
	override_background = {
		reference = pam_the_christian_church_instability 
	}

	widgets = {
		widget = {
 			gui = "event_window_widget_situation_info_christian_church"
 			container = "dynamic_content_widget"
 			controller = situation_info
 			setup_scope = { situation:the_christian_church = { save_scope_as = situation } }
		}
	}

	immediate = {
		play_music_cue = mx_cue_the_shepherds_flock #AUDIO cue track
		save_scope_value_as = {
			name = start
			value = yes
		}	
		save_christian_church_scopes_effect = yes
	}
	option = {
		name = {
			trigger = {
				has_religion = religion:christianity_religion
			}
			text = pam_the_christian_church.0051.a
		}
		name = {
			trigger = {
				NOT = {
					has_religion = religion:christianity_religion
				}
			}
			text = pam_the_christian_church.0051.b
		}				
		clicksound = "event:/DLC/FP2/SFX/UI/fp2_struggle_start_select"
	}
}
