namespace = pam_kingdom_of_heaven

# Kingdom of Heaven — big-screen proclamation (effects in immediate)
pam_kingdom_of_heaven.0001 = {
	type = character_event
	window = big_event_window
	theme = christian
	title = pam_kingdom_of_heaven.0001.t
	desc = pam_kingdom_of_heaven.0001.desc
	override_background = { reference = holy_site_jerusalem } # TODO_PAM_VP
	override_effect_2d = {
		reference = legend_glow
	}

	left_portrait = {
		character = root
		animation = acknowledging_crozier
	}

	immediate = {
		play_music_cue = "mx_cue_crusade_starts"
		capital_province = { save_scope_as = previous_capital }
		pam_kingdom_of_heaven_declaration_effect = yes
	}

	option = { # Move the capital to previous capital
		name = pam_kingdom_of_heaven.0001.a
		set_realm_capital = scope:previous_capital.county
		hidden_effect = {
			domicile ?= { move_domicile = scope:previous_capital }
		}
		ai_chance = { base = 0 }
	}
	
	option = { # Keep the capital where it is
		name = pam_kingdom_of_heaven.0001.b
		ai_chance = { base = 100 }
	}
	
	after = {
		every_player = {
			limit = {
				NOT = { this = scope:kingdom_of_heaven_founder }
				faith = scope:kingdom_of_heaven_founder.faith
			}
			send_interface_toast = {
				title = pam_kingdom_of_heaven_news_title
				custom_tooltip = pam_kingdom_of_heaven_news_tt
				left_icon = scope:kingdom_of_heaven_founder
				right_icon = title:h_kingdom_of_heaven
			}
		}
	}
}
