namespace = pam_antipope_events

pam_antipope_events.0001 = {
	type = character_event
	title = pam_antipope_events.0001.t
	desc = {
		desc = pam_antipope_events.0001.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = {
					this = scope:antipope
				}
				desc = pam_antipope_events.0001.desc.self
			}
			desc = pam_antipope_events.0001.desc.sponsor
		}
	}
	theme = christian
	left_portrait = {
		character = root
		triggered_animation = {
			trigger = { this = scope:antipope }
			animation = preaching
		}
		animation = personality_zealous
	}
	right_portrait = {
		trigger = {
			NOT = { this = root }
		}
		character = scope:antipope
		animation = preaching
	}
	lower_center_portrait = {
		character = scope:pope
		animation = anger
	}

	immediate = {
		save_scope_as = antipope_declarer
		if = {
			limit = { NOT = { exists = scope:antipope } }
			save_scope_as = antipope
			save_scope_as = antipope_self
			pam_establish_antipope_effect = {
				ANTIPOPE = root
				SPONSOR = root
			}
			scope:pope = {
				if = {
					limit = { can_set_relation_rival_trigger = { CHARACTER = root } }
					set_relation_rival = {
						target = root
						reason = rival_declared_antipope
					}
				}
			}
		}
		else = {
			pam_establish_antipope_effect = {
				ANTIPOPE = scope:antipope
				SPONSOR = root
			}
			scope:pope = {
				if = {
					limit = { can_set_relation_rival_trigger = { CHARACTER = scope:antipope } }
					set_relation_rival = {
						target = scope:antipope
						reason = rival_declared_antipope
					}
				}
				add_opinion = {
					opinion = -25
					target = root
					modifier = angry_opinion
				}
			}
		}
	}

	option = {
		name = pam_antipope_events.0001.a
		show_as_tooltip = {
			scope:antipope = { get_title = scope:new_title }
		}
		hidden_effect = {
			every_player = {
				limit = {
					faith.religion = religion:christianity_religion
					NOR = {
						this = scope:sponsor
						this = scope:antipope
					}
				}
				trigger_event = {
					id = pam_antipope_events.0002
					days = 1
				}
			}
		}
	}
}

scripted_effect pam_challenge_hof_war_judge_imprison_effect = {
	rightfully_imprison_character_less_verbose_effect = {
		TARGET = $TARGET$
		IMPRISONER = root
	}
	$TARGET$ = {
		set_variable = {
			name = pam_ch_hof_fate
			value = flag:imprisoned
			days = 30
		}
	}
}

scripted_trigger pam_challenge_hof_war_torture_harm_possible_trigger = {
	OR = {
		NOT = { has_trait = blind }
		NOT = { has_trait = disfigured }
		AND = {
			is_eunuch_trigger = no
			is_male = yes
		}
		NOT = { has_trait = maimed }
		NOT = { has_trait = scarred }
	}
}

# $VICTIM$: a Sadistic or Torturer judge can use this while harm remains, in place of imprison
scripted_trigger pam_challenge_hof_war_judge_torture_option_available_trigger = {
	OR = {
		has_trait = sadistic
		has_trait = torturer
	}
	$VICTIM$ = {
		pam_challenge_hof_war_torture_harm_possible_trigger = yes
	}
}

# $TARGET$: we excommunicate, imprison, and add one random lasting injury, but they do not die
scripted_effect pam_challenge_hof_war_judge_torture_harm_effect = {
	pam_challenge_hof_war_judge_excommunicate_effect = { TARGET = $TARGET$ }
	pam_challenge_hof_war_judge_imprison_effect = { TARGET = $TARGET$ }
	reverse_add_opinion = {
		target = $TARGET$
		modifier = tortured_me
	}
	$TARGET$ = {
		set_variable = {
			name = pam_ch_hof_fate
			value = flag:tortured
			days = 30
		}
		add_character_modifier = {
			modifier = recently_tortured
			years = 5
		}
		custom_tooltip = pam_challenge_hof_war_random_permanent_harm_tt
		hidden_effect = {
			random_list = {
				1 = {
					trigger = { NOT = { has_trait = blind } }
					add_trait = blind
				}
				1 = {
					trigger = { NOT = { has_trait = disfigured } }
					add_trait = disfigured
				}
				1 = {
					trigger = {
						is_eunuch_trigger = no
						is_male = yes
					}
					add_eunuch_trait_effect = yes
				}
				1 = {
					trigger = { NOT = { has_trait = maimed } }
					apply_maimed_trait_and_modifier_effect = yes
				}
				1 = {
					trigger = { NOT = { has_trait = scarred } }
					add_trait = scarred
				}
			}
		}
	}
}

scripted_effect pam_challenge_hof_war_judge_excommunicate_effect = {
	pam_challenge_hof_war_excommunicate_if_possible_effect = { TARGET = $TARGET$ }
}

scripted_effect pam_challenge_hof_war_judge_mercy_effect = {
	$TARGET$ = {
		set_variable = {
			name = pam_ch_hof_fate
			value = flag:mercy
			days = 30
		}
		add_opinion = {
			target = root
			modifier = grateful_opinion
			opinion = 30
		}
	}
	add_piety = medium_piety_gain
	pam_challenge_hof_war_notify_victim_fate_effect = {
		VICTIM = $TARGET$
		JUDGE = root
	}
}

scripted_effect pam_challenge_hof_war_chain_sponsor_judgment_or_followup_effect = {
	if = {
		limit = {
			exists = scope:pam_ch_hof_sponsor
			exists = scope:pam_ch_hof_challenger
			NOT = { scope:pam_ch_hof_sponsor = scope:pam_ch_hof_challenger }
			scope:pam_ch_hof_sponsor = { is_alive = yes }
		}
		trigger_event = {
			id = pam_antipope_events.0011
			days = 1
		}
	}
	else = {
		pam_challenge_hof_war_fire_followup_events_effect = yes
	}
}

		override_imprisonment_visuals = yes
	}
	right_portrait = {
		trigger = { exists = scope:pam_ch_hof_judge }
		character = scope:pam_ch_hof_judge
		triggered_animation = {
			trigger = {
				root = { var:pam_ch_hof_fate = flag:tortured }
			}
			animation = whipping
		}
		animation = anger
	}

	immediate = {
		var:pam_ch_hof_judge ?= { save_scope_as = pam_ch_hof_judge }
		if = {
			limit = { NOT = { exists = scope:pam_ch_hof_judge } }
			var:pam_ch_hof_executioner ?= { save_scope_as = pam_ch_hof_judge }
		}
	}

	# Facing execution: let the house claim you as a martyr
	option = {
		name = pam_antipope_events.0014.a
		trigger = { var:pam_ch_hof_fate = flag:executed }
		house ?= {
			add_house_modifier = {
				modifier = pam_house_martyred_pontiff_modifier
				years = 20
			}
		}
	}

	# Facing execution: leave the judge carrying the guilt of your blood
	option = {
		name = pam_antipope_events.0014.b
		trigger = { var:pam_ch_hof_fate = flag:executed }
		scope:pam_ch_hof_judge ?= {
			add_character_modifier = {
				modifier = pam_ch_hof_sacred_bloodguilt_modifier
				years = 20
			}
			# The judge never sees this event, so tell them about it if they are a player
			hidden_effect = {
				if = {
					limit = { is_ai = no }
					send_interface_toast = {
						type = event_toast_effect_bad
						title = pam_antipope_events.0014.b.toast
						left_icon = root
						show_as_tooltip = {
							add_character_modifier = {
								modifier = pam_ch_hof_sacred_bloodguilt_modifier
								years = 20
							}
						}
					}
				}
			}
		}
	}

	# Facing execution, last words for the family: we clear the house stain, or prestige to heir or self
	option = {
		name = pam_antipope_events.0014.c
		trigger = { var:pam_ch_hof_fate = flag:executed }
		if = {
			limit = {
				exists = house
				house = { has_house_modifier = pam_house_schismatic_sponsor_modifier }
			}
			house = { remove_house_modifier = pam_house_schismatic_sponsor_modifier }
		}
		else_if = {
			limit = { exists = player_heir }
			player_heir = { add_prestige = medium_prestige_gain }
		}
		else = {
			add_prestige = medium_prestige_gain
		}
	}

	# Spared: clear the house's stain if it carries one, and gain piety either way
	option = {
		name = pam_antipope_events.0014.d
		trigger = { NOT = { var:pam_ch_hof_fate = flag:executed } }
		if = {
			limit = {
				exists = house
				house = { has_house_modifier = pam_house_schismatic_sponsor_modifier }
			}
			house = { remove_house_modifier = pam_house_schismatic_sponsor_modifier }
		}
		add_piety = medium_piety_gain
	}

	# Spared: start plotting the judge's murder, or just learn from the humiliation
	option = {
		name = pam_antipope_events.0014.e
		trigger = { NOT = { var:pam_ch_hof_fate = flag:executed } }
		if = {
			limit = {
				exists = scope:pam_ch_hof_judge
				can_start_scheme = {
					type = murder
					target_character = scope:pam_ch_hof_judge
				}
			}
			start_scheme = {
				type = murder
				target_character = scope:pam_ch_hof_judge
			}
			custom_tooltip = pam_antipope_events.0014.e.tt
			hidden_effect = {
				random_scheme = {
					limit = {
						scheme_type = murder
						scheme_target_character = scope:pam_ch_hof_judge
					}
					add_scheme_modifier = {
						type = extra_success_chance_modifier
					}
				}
			}
		}
		else = {
			add_intrigue_lifestyle_xp = medium_lifestyle_xp
		}
	}

	# Spared: swear to do penance for what you attempted
	option = {
		name = pam_antipope_events.0014.f
		trigger = { NOT = { var:pam_ch_hof_fate = flag:executed } }
		add_character_modifier = {
			modifier = pam_ch_hof_penitent_road_modifier
			years = 10
		}
	}

	after = {
		if = {
			limit = { var:pam_ch_hof_fate = flag:executed }
			death = {
				killer = scope:pam_ch_hof_judge
				death_reason = death_execution
			}
		}
	}
