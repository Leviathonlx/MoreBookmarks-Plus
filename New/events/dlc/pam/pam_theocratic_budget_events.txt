namespace = pam_theocratic_budget_events

###########################
## pam_theocratic_budget_events.0001 - Yearly budget enact (maintenance)
###########################

# Fired from yearly_playable_pulse, which already guarantees top liege + has_treasury
pam_theocratic_budget_events.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		this = top_liege
		has_treasury = yes
		government_has_mechanic = theocracy
		OR = {
			treasury_months_since_budget_enact_date >= 12 # yearly (might have had a manual enacting recently)
			monthly_character_treasury_balance < -50 # do it anyway if they have a big deficit
		}
	}

	immediate = {
		enact_treasury_budgets_no_costs = yes
	}
}
