petition_head_of_faith_start = {
	effect = {
		petition_head_of_faith_activate_pending_effect = yes
	}
}

petition_head_of_faith_travel_planner_exit = {
	trigger = {
		has_variable = pending_petition_head_of_faith_recipient
	}
	effect = {
		if = {
			limit = {
				has_variable = pending_petition_head_of_faith_reserved_piety_value
			}
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = petition_head_of_faith_costs_refunded
				desc = petition_head_of_faith_costs_refunded
				left_icon = root
				petition_head_of_faith_refund_pending_costs_effect = yes
			}
		}

		petition_head_of_faith_clear_pending_effect = yes

		# Cancelled after departure: refund the activated reservation and clean up.
		# 0601 may already have refunded in immediate before option A cancels travel;
		# in that case petition_head_of_faith_reserved_piety_value is gone and this skips.
		if = {
			limit = { has_variable = petition_head_of_faith_reserved_piety_value }
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = petition_head_of_faith_costs_refunded
				desc = petition_head_of_faith_costs_refunded
				left_icon = root
				petition_head_of_faith_refund_reserved_costs_effect = yes
			}
		}
		if = {
			limit = { has_character_flag = petition_head_of_faith_character_flag }
			petition_head_of_faith_cleanup_effect = yes
		}
	}
}
