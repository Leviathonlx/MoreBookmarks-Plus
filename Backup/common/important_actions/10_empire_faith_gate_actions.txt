action_efg_request_excommunication = {
	priority = 950
	combine_into_one = yes
	is_dangerous = yes
	type = action
	icon = action_take_decision

	check_create_action = {
		if = {
			limit = {
				is_ai = no
				has_contested_empire_trigger = yes
				faith_has_existing_empire_holder_trigger = yes
				petition_head_of_faith_has_religious_authority_trigger = yes
				NOT = {
					petition_head_of_faith_religious_authority_is_trigger = { CHARACTER = root }
				}
				any_independent_ruler = {
					NOT = { this = root }
					is_alive = yes
					is_landed = yes
					highest_held_title_tier >= tier_empire
					government_subject_to_faith_empire_trigger = yes
					faith = root.faith
					NOT = { has_trait = excommunicated }
					age >= 12
					petition_head_of_faith_excommunication_target_eligible_trigger = { ACTOR = root }
				}
			}
			petition_head_of_faith_save_religious_authority_effect = { SCOPE = petition_religious_authority }

			every_independent_ruler = {
				limit = {
					NOT = { this = root }
					is_alive = yes
					is_landed = yes
					highest_held_title_tier >= tier_empire
					government_subject_to_faith_empire_trigger = yes
					faith = root.faith
					NOT = { has_trait = excommunicated }
					age >= 12
					petition_head_of_faith_excommunication_target_eligible_trigger = { ACTOR = root }
				}
				try_create_important_action = {
					important_action_type = action_efg_request_excommunication
					actor = root
					recipient = this
					secondary_recipient = scope:petition_religious_authority
				}
			}
		}
	}

	effect = {
		# Players who can petition travel to the Head of Faith
		if = {
			limit = {
				has_pam_dlc_trigger = yes
				scope:actor = {
					faith = { petition_head_of_faith_faith_qualifies_trigger = yes }
				}
			}
			open_interaction_window = {
				interaction = petition_head_of_faith_excommunication_setup_interaction
				actor = scope:actor
				recipient = scope:secondary_recipient
				secondary_recipient = scope:recipient
			}
		}
		else = {
			open_interaction_window = {
				interaction = request_excommunication_interaction
				actor = scope:actor
				recipient = scope:secondary_recipient
				secondary_recipient = scope:recipient
			}
		}
	}
}
