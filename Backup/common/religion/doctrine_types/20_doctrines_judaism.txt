# Tracks the way in which Halakha is supposed to be interpreted
temple_authority_rabbinic_doctrine = { # Rabbis argue and interpret the Halakha
	doctrine_group_type = doctrine_temple_authority
	index = 0
	divergence = 50

	icon = "temple_authority_rabbinic_doctrine"

	piety_cost = {
		if = {
			limit = { rite_has_doctrine = temple_authority_rabbinic_doctrine }
			value = 0
		}
		else_if = {
			limit = { rite_has_doctrine = temple_authority_samaritan_torah_doctrine }
			value = faith_tenet_cost_mid
		}
		else = {
			value = faith_tenet_cost_low
		}
	}

	character_modifier = {
		clergy_opinion = 5
		church_holding_build_gold_cost = -0.1
	}

	parameters = {
		temple_authority_rabbinic_doctrine
	}

	special_parameters = {
		hostility_level_override = {
			temple_authority_rabbinic_doctrine = astray
			temple_authority_samaritan_torah_doctrine = evil
		}
	}
}
