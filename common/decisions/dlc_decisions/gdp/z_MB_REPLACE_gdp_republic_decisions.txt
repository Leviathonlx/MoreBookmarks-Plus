# gdp_restore_roman_republic_decision — replaces the vanilla decision in dlc_decisions/gdp/gdp_republic_decisions.txt
gdp_restore_roman_republic_decision = {
	title = gdp_restore_roman_republic_decision
	desc = gdp_restore_roman_republic_decision_desc
	picture = {
		reference = "gfx/interface/illustrations/decisions/ep3_decision_roman_restoration.dds"
	}

	sort_order = 105
	decision_group_type = major

	cost = {
	}

	is_shown = {
		has_gdp_dlc_trigger = yes
		government_has_flag = government_is_republic
		# Must have a culture that somehow descends from Rome
		culture = {
			OR = {
				has_cultural_tradition = tradition_republican_legacy
				has_cultural_pillar = heritage_latin
				any_parent_culture_or_above = {
					this = culture:roman
				}
			}
		}
		# You may only restore it once
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_restored_roman_republic
			}
		}
	}

	is_valid = {
		is_independent_ruler = yes
		has_realm_law = republic_tier_empire
		prestige_level >= max_prestige_level
		completely_controls = title:d_latium
		completely_controls = title:d_pisa
		completely_controls = title:d_toscana
		completely_controls = title:d_ancona
		completely_controls = title:d_spoleto
		completely_controls = title:d_benevento
		completely_controls = title:d_capua
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		NOT = { exists = title:h_roman_empire.holder }
	}

	effect = {
		show_as_tooltip = { # Actual effect is applied in follow-up event
			gdp_restore_the_roman_republic_effect = yes
		}
		if = {
			limit = {
				capital_county != title:c_roma_urbs
			}
			custom_tooltip = restore_roman_republic_make_rome_capital_desc
		}
		trigger_event = republic_event.0100

		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_restored_roman_republic
			}
			add_achievement_global_variable_effect = {
				VARIABLE = ep5_06_senatus_populusque_romanus_achievement_unlocked
				VALUE = yes
			}
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 36
		hegemony = 0
	}

	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}
