almaden_mines_01 = {
	effect_desc = mines_effect_desc

	construction_time = slow_construction_time

	type_icon = "special_buildings/mines.dds"

	asset = {
		type = pdxmesh
		name = "building_special_mines_mesh"
	}

	can_construct_potential = {
		building_requirement_tribal = no
	}

	can_construct = {
		building_requirement_castle_city_church = { LEVEL = 01 }
	}
	
	cost_gold = expensive_building_tier_3_cost
	
	province_modifier = {
		monthly_mining_tax = special_mine_building_tax_tier_1
	}
	county_modifier = {
		trade_good_lead_production_mult = special_mine_tier_1_mult
	}
	
	next_building = almaden_mines_02
	
	market_modifier = {
		trade_good_wood_sell_price_mult = special_building_need_sell_price
	}

	ai_value = {
		base = 10
		directive_to_build_economy_modifier = yes
		modifier = { # Fill all building slots before going for special buildings
			factor = 0
			free_building_slots > 0
		}
		ai_special_district_modifier = { LEVEL = 1 }
	}
	
	
	trade_good_need = {
		trade_good = trade_good_wood
		amount = special_building_need_amount
		can_be_filled_multiple_times = 5

		modifiers = {
			county_modifier = {
				prosperity_growth_factor = special_building_need_prosperity_minimal
			}
		}
	}

	flag = travel_point_of_interest_economic
	flag = special_mine
}
